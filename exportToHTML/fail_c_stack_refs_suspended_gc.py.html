fail_c_stack_refs_suspended_gc.py 
1    """Regression test for issue #515: GC of a suspended greenlet's C-stack refs. 
2     
3    A greenlet that suspends mid attribute-resolution holds a ``_PyCStackRef`` to 
4    the looked-up object on its C stack. The free-threaded collector only walks the 
5    running thread's refs, so greenlet must visit a suspended greenlet's from 
6    ``tp_traverse``, or an object reachable only through them is freed early. 
7     
8    The test pins a deferred-refcounted class (a metaclass ``__get__`` makes the 
9    class act as a descriptor), switches away from inside ``__get__``, drops every 
10   other reference, and collects. It passes only if the class is still alive and 
11   the suspended greenlet is what keeps it alive. Prints "C STACK REFS GC OK" and 
12   exits 0 on a fixed build, non-zero on a regressed one. 
13   """
14   import gc
15   import sys
16   import sysconfig
17   import weakref
18   
19   import greenlet
20   
21   # Only free-threaded builds have the C-stack-ref / deferred-refcount machinery.
22   FREETHREAD = bool(sysconfig.get_config_var("Py_GIL_DISABLED"))
23   
24   parent = greenlet.getcurrent()
25   observed = {}
26   
27   
28   class Meta(type):
29       def __get__(cls, obj, objtype=None):
30           # ``cls`` is the class being resolved, pinned in a _PyCStackRef across
31           # this call. Drop the descriptor-protocol locals so a suspended frame
32           # can't keep the class alive on its own and mask the bug.
33           del cls, obj, objtype
34           child.switch()
35           return 42
36   
37   
38   pinned = Meta('pinned', (), {})                    # a class => deferred-refcounted
39   holder = type('holder', (), {'attr': pinned})      # holder.attr invokes Meta.__get__
40   box = [pinned]
41   del pinned
42   
43   
44   def child_work():
45       # parent is suspended inside Meta.__get__ holding a C-stack ref to the class.
46       # Drop every other reference, then collect.
47       ref = weakref.ref(box[0])
48       box[0] = None
49       del holder.attr
50       for _ in range(5):
51           gc.collect()
52       cls = ref()
53       if cls is None:
54           observed['status'] = 'collected'
55       elif not FREETHREAD or any(r is parent for r in gc.get_referrers(cls)):
56           # Kept alive by the suspended greenlet (or, with the GIL, by refcounting).
57           observed['status'] = 'ok'
58       else:
59           # Alive, but not because of the greenlet: a masking reference hid the
60           # C-stack-ref path (the class's own mro/bases are a self-cycle, not one).
61           observed['status'] = 'masked'
62           own = (cls.__mro__, cls.__bases__)
63           observed['maskers'] = sorted(
64               {type(r).__name__ for r in gc.get_referrers(cls)
65                if r is not parent and r is not own[0] and r is not own[1]}
66           ) or ['<invisible to gc; strong C-stack ref / non-deferred class>']
67       del cls
68       parent.switch()
69   
70   
71   child = greenlet.greenlet(child_work)
72   result = holder().attr
73   assert result == 42, result
74   
75   status = observed.get('status')
76   print(f"py={sys.version.split()[0]} gil={getattr(sys, '_is_gil_enabled', lambda: True)()} "
77         f"greenlet={greenlet.__version__} status={status}", flush=True)
78   if status == 'collected':
79       raise SystemExit("REGRESSED: class reachable only through a suspended "
80                        "greenlet's C-stack ref was collected early")
81   if status == 'masked':
82       raise SystemExit("REGRESSED: class stayed alive but not through the suspended "
83                        "greenlet; a masking reference (%s) hid the C-stack-ref path"
84                        % ', '.join(observed.get('maskers', ())))
85   print("C STACK REFS GC OK", flush=True)
