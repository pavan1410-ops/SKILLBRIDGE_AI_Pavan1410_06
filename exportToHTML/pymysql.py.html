24   SSL Connections 
25   ------------------ 
26    
27   The PyMySQL DBAPI accepts the same SSL arguments as that of MySQLdb, 
28   described at :ref:`mysqldb_ssl`.   See that section for additional examples. 
29    
30   If the server uses an automatically-generated certificate that is self-signed 
31   or does not match the host name (as seen from the client), it may also be 
32   necessary to indicate ``ssl_check_hostname=false`` in PyMySQL:: 
33    
34       connection_uri = ( 
35           "mysql+pymysql://scott:tiger@192.168.0.134/test" 
36           "?ssl_ca=/home/gord/client-ssl/ca.pem" 
37           "&ssl_cert=/home/gord/client-ssl/client-cert.pem" 
38           "&ssl_key=/home/gord/client-ssl/client-key.pem" 
39           "&ssl_check_hostname=false" 
40       )

74   def _connection_ping_reconnects_true(connection_cls: Type[Any]) -> bool:
75       """Given a Connection class like pymysql.Connection, aiomysql.Connection, 
76       asyncmy.Connection, inspect the ping() method and determine if it 
77       has a "reconnect" parameter that either defaults to True, or is positional. 
78    
79       a return value of True here means that when we call ``connection.ping()``, 
80       we **must** pass `reconnect=False`.  a return value of False means that 
81       we should call ``connection.ping()`` with **no arguments**. 
82    
83       This routine originates from issue #10492 for pymysql, however arg 
84       signature mismatches in aiomysql/asyncmy tracked by issue #13306 
85       necessitated a more open ended function. 
86    
87       """
88       insp = langhelpers.get_callable_argspec(connection_cls.ping)
89       try:
90           reconnect_arg = insp.args[1]
91       except IndexError:
92           return False
93       else:
94           return reconnect_arg == "reconnect" and (
95               not insp.defaults or insp.defaults[0] is not False
96           )

99   class MySQLDialect_pymysql(MySQLDialect_mysqldb):
100      driver = "pymysql"
101      supports_statement_cache = True
102  
103      description_encoding = None
104  
105      @langhelpers.memoized_property
106      def supports_server_side_cursors(self) -> bool:
107          try:
108              cursors = __import__("pymysql.cursors").cursors
109              self._sscursor = cursors.SSCursor
110              return True
111          except (ImportError, AttributeError):
112              return False
113  
114      @classmethod
115      def import_dbapi(cls) -> DBAPIModule:
116          return __import__("pymysql")
117  
118      @langhelpers.memoized_property
119      def _send_false_to_ping(self) -> bool:
120          """determine if pymysql has deprecated, changed the default of, 
121          or removed the 'reconnect' argument of connection.ping(). 
122   
123          See #10492 and 
124          https://github.com/PyMySQL/mysqlclient/discussions/651#discussioncomment-7308971 
125          for background. 
126   
127          Revised as part of #13306 
128   
129          """  # noqa: E501
130  
131          try:
132              Connection = __import__(
133                  "pymysql.connections"
134              ).connections.Connection
135          except (ImportError, AttributeError):
136              return True
137          else:
138              return _connection_ping_reconnects_true(Connection)
139  
140      def do_ping(self, dbapi_connection: DBAPIConnection) -> Literal[True]:
141          if self._send_false_to_ping:
142              dbapi_connection.ping(False)
143          else:
144              dbapi_connection.ping()
145  
146          return True
147  
148      def create_connect_args(
149          self, url: URL, _translate_args: Optional[Dict[str, Any]] = None
150      ) -> ConnectArgsType:
151          if _translate_args is None:
152              _translate_args = dict(username="user")
153          return super().create_connect_args(
154              url, _translate_args=_translate_args
155          )
156  
157      def is_disconnect(
158          self,
159          e: DBAPIModule.Error,
160          connection: Optional[Union[PoolProxiedConnection, DBAPIConnection]],
161          cursor: Optional[DBAPICursor],
162      ) -> bool:
163          if super().is_disconnect(e, connection, cursor):
164              return True
165          elif isinstance(e, self.loaded_dbapi.Error):
166              str_e = str(e).lower()
167              return (
168                  "already closed" in str_e or "connection was killed" in str_e
169              )
170          else:
171              return False
172  
173      def _extract_error_code(self, exception: BaseException) -> Any:
174          if isinstance(exception.args[0], Exception):
175              exception = exception.args[0]
176          return exception.args[0]
