28   def encode_cookie(payload, key=None):
29       """ 
30       This will encode a ``str`` value into a cookie, and sign that cookie 
31       with the app's secret key. 
32    
33       :param payload: The value to encode, as `str`. 
34       :type payload: str 
35    
36       :param key: The key to use when creating the cookie digest. If not 
37                   specified, the SECRET_KEY value from app config will be used. 
38       :type key: str 
39       """
40       return f"{payload}|{_cookie_digest(payload, key=key)}"

43   def decode_cookie(cookie, key=None):
44       """ 
45       This decodes a cookie given by `encode_cookie`. If verification of the 
46       cookie fails, ``None`` will be implicitly returned. 
47    
48       :param cookie: An encoded cookie. 
49       :type cookie: str 
50    
51       :param key: The key to use when creating the cookie digest. If not 
52                   specified, the SECRET_KEY value from app config will be used. 
53       :type key: str 
54       """
55       try:
56           payload, digest = cookie.rsplit("|", 1)
57           if hasattr(digest, "decode"):
58               digest = digest.decode("ascii")  # pragma: no cover
59       except ValueError:
60           return
61   
62       if hmac.compare_digest(_cookie_digest(payload, key=key), digest):
63           return payload

66   def make_next_param(login_url, current_url):
67       """ 
68       Reduces the scheme and host from a given URL so it can be passed to 
69       the given `login` URL more efficiently. 
70    
71       :param login_url: The login URL being redirected to. 
72       :type login_url: str 
73       :param current_url: The URL to reduce. 
74       :type current_url: str 
75       """
76       l_url = urlsplit(login_url)
77       c_url = urlsplit(current_url)
78   
79       if (not l_url.scheme or l_url.scheme == c_url.scheme) and (
80           not l_url.netloc or l_url.netloc == c_url.netloc
81       ):
82           return urlunsplit(("", "", c_url.path, c_url.query, ""))
83       return current_url

86   def expand_login_view(login_view):
87       """ 
88       Returns the url for the login view, expanding the view name to a url if 
89       needed. 
90    
91       :param login_view: The name of the login view or a URL for the login view. 
92       :type login_view: str 
93       """
94       if login_view.startswith(("https://", "http://", "/")):
95           return login_view
96   
97       return url_for(login_view)

100  def login_url(login_view, next_url=None, next_field="next"):
101      """ 
102      Creates a URL for redirecting to a login page. If only `login_view` is 
103      provided, this will just return the URL for it. If `next_url` is provided, 
104      however, this will append a ``next=URL`` parameter to the query string 
105      so that the login view can redirect back to that URL. Flask-Login's default 
106      unauthorized handler uses this function when redirecting to your login url. 
107      To force the host name used, set `FORCE_HOST_FOR_REDIRECTS` to a host. This 
108      prevents from redirecting to external sites if request headers Host or 
109      X-Forwarded-For are present. 
110   
111      :param login_view: The name of the login view. (Alternately, the actual 
112                         URL to the login view.) 
113      :type login_view: str 
114      :param next_url: The URL to give the login view for redirection. 
115      :type next_url: str 
116      :param next_field: What field to store the next URL in. (It defaults to 
117                         ``next``.) 
118      :type next_field: str 
119      """
120      base = expand_login_view(login_view)
121  
122      if next_url is None:
123          return base
124  
125      parsed_result = urlsplit(base)
126      md = parse_qs(parsed_result.query, keep_blank_values=True)
127      md[next_field] = make_next_param(base, next_url)
128      netloc = current_app.config.get("FORCE_HOST_FOR_REDIRECTS") or parsed_result.netloc
129      parsed_result = parsed_result._replace(
130          netloc=netloc, query=urlencode(md, doseq=True)
131      )
132      return urlunsplit(parsed_result)

142  def login_remembered():
143      """ 
144      This returns ``True`` if the current login is remembered across sessions. 
145      """
146      config = current_app.config
147      cookie_name = config.get("REMEMBER_COOKIE_NAME", COOKIE_NAME)
148      has_cookie = cookie_name in request.cookies and session.get("_remember") != "clear"
149      if has_cookie:
150          cookie = request.cookies[cookie_name]
151          user_id = decode_cookie(cookie)
152          return user_id is not None
153      return False

156  def login_user(user, remember=False, duration=None, force=False, fresh=True):
157      """ 
158      Logs a user in. You should pass the actual user object to this. If the 
159      user's `is_active` property is ``False``, they will not be logged in 
160      unless `force` is ``True``. 
161   
162      This will return ``True`` if the log in attempt succeeds, and ``False`` if 
163      it fails (i.e. because the user is inactive). 
164   
165      :param user: The user object to log in. 
166      :type user: object 
167      :param remember: Whether to remember the user after their session expires. 
168          Defaults to ``False``. 
169      :type remember: bool 
170      :param duration: The amount of time before the remember cookie expires. If 
171          ``None`` the value set in the settings is used. Defaults to ``None``. 
172      :type duration: :class:`datetime.timedelta` 
173      :param force: If the user is inactive, setting this to ``True`` will log 
174          them in regardless. Defaults to ``False``. 
175      :type force: bool 
176      :param fresh: setting this to ``False`` will log in the user with a session 
177          marked as not "fresh". Defaults to ``True``. 
178      :type fresh: bool 
179      """
180      if not force and not user.is_active:
181          return False
182  
183      user_id = getattr(user, current_app.login_manager.id_attribute)()
184      session["_user_id"] = user_id
185      session["_fresh"] = fresh
186      session["_id"] = current_app.login_manager._session_identifier_generator()
187  
188      if remember:
189          session["_remember"] = "set"
190          if duration is not None:
191              try:
192                  # equal to timedelta.total_seconds() but works with Python 2.6
193                  session["_remember_seconds"] = (
194                      duration.microseconds
195                      + (duration.seconds + duration.days * 24 * 3600) * 10**6
196                  ) / 10.0**6
197              except AttributeError as e:
198                  raise Exception(
199                      f"duration must be a datetime.timedelta, instead got: {duration}"
200                  ) from e
201  
202      current_app.login_manager._update_request_context_with_user(user)
203      user_logged_in.send(current_app._get_current_object(), user=_get_user())
204      return True

207  def logout_user():
208      """ 
209      Logs a user out. (You do not need to pass the actual user.) This will 
210      also clean up the remember me cookie if it exists. 
211      """
212  
213      user = _get_user()
214  
215      if "_user_id" in session:
216          session.pop("_user_id")
217  
218      if "_fresh" in session:
219          session.pop("_fresh")
220  
221      if "_id" in session:
222          session.pop("_id")
223  
224      cookie_name = current_app.config.get("REMEMBER_COOKIE_NAME", COOKIE_NAME)
225      if cookie_name in request.cookies:
226          session["_remember"] = "clear"
227          if "_remember_seconds" in session:
228              session.pop("_remember_seconds")
229  
230      user_logged_out.send(current_app._get_current_object(), user=user)
231  
232      current_app.login_manager._update_request_context_with_user()
233      return True

246  def login_required(func):
247      """ 
248      If you decorate a view with this, it will ensure that the current user is 
249      logged in and authenticated before calling the actual view. (If they are 
250      not, it calls the :attr:`LoginManager.unauthorized` callback.) For 
251      example:: 
252   
253          @app.route('/post') 
254          @login_required 
255          def post(): 
256              pass 
257   
258      If there are only certain times you need to require that your user is 
259      logged in, you can do so with:: 
260   
261          if not current_user.is_authenticated: 
262              return current_app.login_manager.unauthorized() 
263   
264      ...which is essentially the code that this function adds to your views. 
265   
266      It can be convenient to globally turn off authentication when unit testing. 
267      To enable this, if the application configuration variable `LOGIN_DISABLED` 
268      is set to `True`, this decorator will be ignored. 
269   
270      .. Note :: 
271   
272          Per `W3 guidelines for CORS preflight requests 
273          <http://www.w3.org/TR/cors/#cross-origin-request-with-preflight-0>`_, 
274          HTTP ``OPTIONS`` requests are exempt from login checks. 
275   
276      :param func: The view function to decorate. 
277      :type func: function 
278      """
279  
280      @wraps(func)
281      def decorated_view(*args, **kwargs):
282          if request.method in EXEMPT_METHODS or current_app.config.get("LOGIN_DISABLED"):
283              pass
284          elif not current_user.is_authenticated:
285              return current_app.login_manager.unauthorized()
286  
287          # flask 1.x compatibility
288          # current_app.ensure_sync is only available in Flask >= 2.0
289          if callable(getattr(current_app, "ensure_sync", None)):
290              return current_app.ensure_sync(func)(*args, **kwargs)
291          return func(*args, **kwargs)
292  
293      return decorated_view

296  def fresh_login_required(func):
297      """ 
298      If you decorate a view with this, it will ensure that the current user's 
299      login is fresh - i.e. their session was not restored from a 'remember me' 
300      cookie. Sensitive operations, like changing a password or e-mail, should 
301      be protected with this, to impede the efforts of cookie thieves. 
302   
303      If the user is not authenticated, :meth:`LoginManager.unauthorized` is 
304      called as normal. If they are authenticated, but their session is not 
305      fresh, it will call :meth:`LoginManager.needs_refresh` instead. (In that 
306      case, you will need to provide a :attr:`LoginManager.refresh_view`.) 
307   
308      Behaves identically to the :func:`login_required` decorator with respect 
309      to configuration variables. 
310   
311      .. Note :: 
312   
313          Per `W3 guidelines for CORS preflight requests 
314          <http://www.w3.org/TR/cors/#cross-origin-request-with-preflight-0>`_, 
315          HTTP ``OPTIONS`` requests are exempt from login checks. 
316   
317      :param func: The view function to decorate. 
318      :type func: function 
319      """
320  
321      @wraps(func)
322      def decorated_view(*args, **kwargs):
323          if request.method in EXEMPT_METHODS or current_app.config.get("LOGIN_DISABLED"):
324              pass
325          elif not current_user.is_authenticated:
326              return current_app.login_manager.unauthorized()
327          elif not login_fresh():
328              return current_app.login_manager.needs_refresh()
329          try:
330              # current_app.ensure_sync available in Flask >= 2.0
331              return current_app.ensure_sync(func)(*args, **kwargs)
332          except AttributeError:  # pragma: no cover
333              return func(*args, **kwargs)
334  
335      return decorated_view

338  def set_login_view(login_view, blueprint=None):
339      """ 
340      Sets the login view for the app or blueprint. If a blueprint is passed, 
341      the login view is set for this blueprint on ``blueprint_login_views``. 
342   
343      :param login_view: The user object to log in. 
344      :type login_view: str 
345      :param blueprint: The blueprint which this login view should be set on. 
346          Defaults to ``None``. 
347      :type blueprint: object 
348      """
349  
350      num_login_views = len(current_app.login_manager.blueprint_login_views)
351      if blueprint is not None or num_login_views != 0:
352          (current_app.login_manager.blueprint_login_views[blueprint.name]) = login_view
353  
354          if (
355              current_app.login_manager.login_view is not None
356              and None not in current_app.login_manager.blueprint_login_views
357          ):
358              (
359                  current_app.login_manager.blueprint_login_views[None]
360              ) = current_app.login_manager.login_view
361  
362          current_app.login_manager.login_view = None
363      else:
364          current_app.login_manager.login_view = login_view

377  def _cookie_digest(payload, key=None):
378      key = _secret_key(key)
379  
380      return hmac.new(key, payload.encode("utf-8"), sha512).hexdigest()

383  def _get_remote_addr():
384      address = request.headers.get("X-Forwarded-For", request.remote_addr)
385      if address is not None:
386          # An 'X-Forwarded-For' header includes a comma separated list of the
387          # addresses, the first address being the actual remote address.
388          address = address.encode("utf-8").split(b",")[0].strip()
389      return address

392  def _create_identifier():
393      user_agent = request.headers.get("User-Agent")
394      if user_agent is not None:
395          user_agent = user_agent.encode("utf-8")
396      base = f"{_get_remote_addr()}|{user_agent}"
397      if str is bytes:
398          base = str(base, "utf-8", errors="replace")  # pragma: no cover
399      h = sha512()
400      h.update(base.encode("utf8"))
401      return h.hexdigest()

408  def _secret_key(key=None):
409      if key is None:
410          key = current_app.config["SECRET_KEY"]
411  
412      if isinstance(key, str):  # pragma: no cover
413          key = key.encode("latin1")  # ensure bytes
414  
415      return key
