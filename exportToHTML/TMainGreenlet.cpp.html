21   // Incremented when we create a main greenlet, in a new thread, decremented
22   // when it is destroyed.
23   #ifdef Py_GIL_DISABLED
24   static std::atomic<Py_ssize_t> G_TOTAL_MAIN_GREENLETS(0);
25   #else
26   // Protected by the GIL.

46   MainGreenlet::MainGreenlet(PyGreenlet* p, ThreadState* state)
47       : Greenlet(p, StackState::make_main()),
48         _self(p),
49         _thread_state(state)
50   {
51       G_TOTAL_MAIN_GREENLETS++;
52   }
53   
54   MainGreenlet::~MainGreenlet()
55   {
56       G_TOTAL_MAIN_GREENLETS--;
57       this->tp_clear();
58   }

94   OwnedObject
95   MainGreenlet::g_switch()
96   {
97       try {
98           this->check_switch_allowed();
99       }
100      catch (const PyErrOccurred&) {
101          this->release_args();
102          throw;
103      }
104  
105      switchstack_result_t err = this->g_switchstack();
106      if (err.status < 0) {
107          // XXX: This code path is untested, but it is shared
108          // with the UserGreenlet path that is tested.
109          return this->on_switchstack_or_initialstub_failure(
110              this,
111              err,
112              true, // target was me
113              false // was initial stub
114          );
115      }
116  
117      return err.the_new_current_greenlet->g_switch_finish(err);
118  }

120  int
121  MainGreenlet::tp_traverse(visitproc visit, void* arg)
122  {
123      ThreadState* thread_state = this->_thread_state.load();
124      if (thread_state) {
125          // we've already traversed main, (self), don't do it again.
126          int result = thread_state->tp_traverse(visit, arg, false);
127          if (result) {
128              return result;
129          }
130      }
131      return Greenlet::tp_traverse(visit, arg);
132  }
