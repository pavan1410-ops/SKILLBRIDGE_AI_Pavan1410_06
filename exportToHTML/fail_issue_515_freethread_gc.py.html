fail_issue_515_freethread_gc.py 
1    """Reproducer for the free-threaded GC crash in issue #515. 
2     
3    https://github.com/python-greenlet/greenlet/issues/515 
4     
5    Before the fix, ``set_initial_state`` copied the parent thread state's 
6    ``_PyCStackRef`` list head into each new greenlet. Those nodes sit on the 
7    parent greenlet's C stack, so once the child starts running on its own stack 
8    and overwrites that region their ``next`` pointers dangle. The free-threaded 
9    collector walks ``c_stack_refs`` for every thread, so a GC on any thread ends 
10   up chasing those dangling nodes (inside ``gc_collect_main`` -> 
11   ``gc_visit_thread_stacks``) while such a child is active, and segfaults. 
12    
13   The fiddly part is getting a *non-empty* list to inherit at all. The 
14   interpreter only holds a ``_PyCStackRef`` transiently, while it is resolving an 
15   attribute -- ``_Py_LoadAttr_StackRefSteal`` pins ``self`` across ``obj.attr()`` 
16   -- so the greenlet has to be primed from inside that window. Priming it from a 
17   property getter does exactly that; it is also the same deep-attribute context 
18   Playwright's sync bridge happens to start its fiber from. A churn thread then 
19   keeps the collector busy while the dispatcher runs on its stale list. 
20    
21   A fixed build prints "ISSUE 515 OK"; a regressed one segfaults in about a 
22   second. With-GIL builds are unaffected -- ordinary refcounting keeps the nodes 
23   alive -- so there it simply prints the OK line. 
24   """
25   import gc
26   import os
27   import sys
28   import threading
29   
30   import greenlet
31   
32   # The crash lands within a second, so a few seconds leaves plenty of margin on
33   # a slow machine without dragging out a fixed build.
34   DURATION = float(os.environ.get("REPRO_SECONDS", "4"))
35   NWORKERS = int(os.environ.get("REPRO_THREADS", "6"))
36   
37   
38   def _busy():
39       # Keep the dispatcher doing work (resolving str methods) so a collection on
40       # the churn thread catches it while it is the active greenlet.
41       return "abc".upper().lower().strip().title()
42   
43   
44   class Bridge:
45       def __init__(self, stop):
46           self.stop = stop
47           self.main = greenlet.getcurrent()
48           self.disp = greenlet.greenlet(self._loop)
49   
50       def _loop(self):
51           while not self.stop[0]:
52               for _ in range(50):
53                   _busy()
54               self.main.switch()
55   
56       @property
57       def _prime(self):
58           # Reaching this getter means we are mid attribute-resolution, so a
59           # _PyCStackRef for ``self`` is live right now. Priming the dispatcher
60           # from here is what makes it inherit a non-empty c_stack_refs head.
61           self.disp.switch()
62           return None
63   
64       def start(self):
65           # Reading the property (rather than calling a method) is deliberate: the
66           # getter runs *during* attribute resolution, so it primes the dispatcher
67           # while the _PyCStackRef for ``self`` is still held. We return the value
68           # only so this isn't a bare, pointless-looking expression statement.
69           return self._prime
70   
71       def resume(self):
72           self.disp.switch()
73   
74   
75   def worker(stop):
76       bridge = Bridge(stop)
77       bridge.start()
78       # start() has returned, so the stack the inherited nodes point at is being
79       # reused -- the dispatcher's list is stale now. Keep it active to trip the GC.
80       while not stop[0]:
81           bridge.resume()
82   
83   
84   def churn(stop):
85       while not stop[0]:
86           # A cycle forces a real (stop-the-world) collection rather than just a
87           # refcount drop.
88           nodes = [{"i": i, "next": None} for i in range(500)]
89           for a, b in zip(nodes, nodes[1:]):
90               a["next"] = b
91           nodes[-1]["next"] = nodes[0]
92           del nodes
93           gc.collect()
94   
95   
96   def main():
97       stop = [False]
98       threads = [threading.Thread(target=churn, args=(stop,), daemon=True)]
99       threads += [threading.Thread(target=worker, args=(stop,), daemon=True)
100                  for _ in range(NWORKERS)]
101      for t in threads:
102          t.start()
103      threading.Event().wait(DURATION)
104      stop[0] = True
105      for t in threads:
106          t.join(timeout=5)
107      print("ISSUE 515 OK")
108  
109  
110  if __name__ == "__main__":
111      main()
112      sys.exit(0)
