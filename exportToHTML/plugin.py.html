60   class SQLAlchemyPlugin(Plugin):
61       def get_dynamic_class_hook(
62           self, fullname: str
63       ) -> Optional[Callable[[DynamicClassDefContext], None]]:
64           if names.type_id_for_fullname(fullname) is names.DECLARATIVE_BASE:
65               return _dynamic_class_hook
66           return None
67   
68       def get_customize_class_mro_hook(
69           self, fullname: str
70       ) -> Optional[Callable[[ClassDefContext], None]]:
71           return _fill_in_decorators
72   
73       def get_class_decorator_hook(
74           self, fullname: str
75       ) -> Optional[Callable[[ClassDefContext], None]]:
76           sym = self.lookup_fully_qualified(fullname)
77   
78           if sym is not None and sym.node is not None:
79               type_id = names.type_id_for_named_node(sym.node)
80               if type_id is names.MAPPED_DECORATOR:
81                   return _cls_decorator_hook
82               elif type_id in (
83                   names.AS_DECLARATIVE,
84                   names.AS_DECLARATIVE_BASE,
85               ):
86                   return _base_cls_decorator_hook
87               elif type_id is names.DECLARATIVE_MIXIN:
88                   return _declarative_mixin_hook
89   
90           return None
91   
92       def get_metaclass_hook(
93           self, fullname: str
94       ) -> Optional[Callable[[ClassDefContext], None]]:
95           if names.type_id_for_fullname(fullname) is names.DECLARATIVE_META:
96               # Set any classes that explicitly have metaclass=DeclarativeMeta
97               # as declarative so the check in `get_base_class_hook()` works
98               return _metaclass_cls_hook
99   
100          return None
101  
102      def get_base_class_hook(
103          self, fullname: str
104      ) -> Optional[Callable[[ClassDefContext], None]]:
105          sym = self.lookup_fully_qualified(fullname)
106  
107          if (
108              sym
109              and isinstance(sym.node, TypeInfo)
110              and util.has_declarative_base(sym.node)
111          ):
112              return _base_cls_hook
113  
114          return None
115  
116      def get_attribute_hook(
117          self, fullname: str
118      ) -> Optional[Callable[[AttributeContext], Type]]:
119          if fullname.startswith(
120              "sqlalchemy.orm.attributes.QueryableAttribute."
121          ):
122              return _queryable_getattr_hook
123  
124          return None
125  
126      def get_additional_deps(
127          self, file: MypyFile
128      ) -> List[Tuple[int, str, int]]:
129          return [
130              #
131              (10, "sqlalchemy.orm", -1),
132              (10, "sqlalchemy.orm.attributes", -1),
133              (10, "sqlalchemy.orm.decl_api", -1),
134          ]

138      return SQLAlchemyPlugin

141  def _dynamic_class_hook(ctx: DynamicClassDefContext) -> None:
142      """Generate a declarative Base class when the declarative_base() function 
143      is encountered."""
144  
145      _add_globals(ctx)
146  
147      cls = ClassDef(ctx.name, Block([]))
148      cls.fullname = ctx.api.qualified_name(ctx.name)
149  
150      info = TypeInfo(SymbolTable(), cls, ctx.api.cur_mod_id)
151      cls.info = info
152      _set_declarative_metaclass(ctx.api, cls)
153  
154      cls_arg = util.get_callexpr_kwarg(ctx.call, "cls", expr_types=(NameExpr,))
155      if cls_arg is not None and isinstance(cls_arg.node, TypeInfo):
156          util.set_is_base(cls_arg.node)
157          decl_class.scan_declarative_assignments_and_apply_types(
158              cls_arg.node.defn, ctx.api, is_mixin_scan=True
159          )
160          info.bases = [Instance(cls_arg.node, [])]
161      else:
162          obj = ctx.api.named_type(names.NAMED_TYPE_BUILTINS_OBJECT)
163  
164          info.bases = [obj]
165  
166      try:
167          calculate_mro(info)
168      except MroError:
169          util.fail(
170              ctx.api, "Not able to calculate MRO for declarative base", ctx.call
171          )
172          obj = ctx.api.named_type(names.NAMED_TYPE_BUILTINS_OBJECT)
173          info.bases = [obj]
174          info.fallback_to_any = True
175  
176      ctx.api.add_symbol_table_node(ctx.name, SymbolTableNode(GDEF, info))
177      util.set_is_base(info)
178

180  def _fill_in_decorators(ctx: ClassDefContext) -> None:
181      for decorator in ctx.cls.decorators:
182          # set the ".fullname" attribute of a class decorator
183          # that is a MemberExpr.   This causes the logic in
184          # semanal.py->apply_class_plugin_hooks to invoke the
185          # get_class_decorator_hook for our "registry.map_class()"
186          # and "registry.as_declarative_base()" methods.
187          # this seems like a bug in mypy that these decorators are otherwise
188          # skipped.
189  
190          if (
191              isinstance(decorator, nodes.CallExpr)
192              and isinstance(decorator.callee, nodes.MemberExpr)
193              and decorator.callee.name == "as_declarative_base"
194          ):
195              target = decorator.callee
196          elif (
197              isinstance(decorator, nodes.MemberExpr)
198              and decorator.name == "mapped"
199          ):
200              target = decorator
201          else:
202              continue
203  
204          if isinstance(target.expr, NameExpr):
205              sym = ctx.api.lookup_qualified(
206                  target.expr.name, target, suppress_errors=True
207              )
208          else:
209              continue
210  
211          if sym and sym.node:
212              sym_type = get_proper_type(sym.type)
213              if isinstance(sym_type, Instance):
214                  target.fullname = f"{sym_type.type.fullname}.{target.name}"
215              else:
216                  # if the registry is in the same file as where the
217                  # decorator is used, it might not have semantic
218                  # symbols applied and we can't get a fully qualified
219                  # name or an inferred type, so we are actually going to
220                  # flag an error in this case that they need to annotate
221                  # it.  The "registry" is declared just
222                  # once (or few times), so they have to just not use
223                  # type inference for its assignment in this one case.
224                  util.fail(
225                      ctx.api,
226                      "Class decorator called %s(), but we can't "
227                      "tell if it's from an ORM registry.  Please "
228                      "annotate the registry assignment, e.g. "
229                      "my_registry: registry = registry()" % target.name,
230                      sym.node,
231                  )

234  def _cls_decorator_hook(ctx: ClassDefContext) -> None:
235      _add_globals(ctx)
236      assert isinstance(ctx.reason, nodes.MemberExpr)
237      expr = ctx.reason.expr
238  
239      assert isinstance(expr, nodes.RefExpr) and isinstance(expr.node, nodes.Var)
240  
241      node_type = get_proper_type(expr.node.type)
242  
243      assert (
244          isinstance(node_type, Instance)
245          and names.type_id_for_named_node(node_type.type) is names.REGISTRY
246      )
247  
248      decl_class.scan_declarative_assignments_and_apply_types(ctx.cls, ctx.api)
249  
250  
251  def _base_cls_decorator_hook(ctx: ClassDefContext) -> None:
252      _add_globals(ctx)
253  
254      cls = ctx.cls
255  
256      _set_declarative_metaclass(ctx.api, cls)
257  
258      util.set_is_base(ctx.cls.info)
259      decl_class.scan_declarative_assignments_and_apply_types(
260          cls, ctx.api, is_mixin_scan=True
261      )
262  
263  
264  def _declarative_mixin_hook(ctx: ClassDefContext) -> None:
265      _add_globals(ctx)
266      util.set_is_base(ctx.cls.info)
267      decl_class.scan_declarative_assignments_and_apply_types(
268          ctx.cls, ctx.api, is_mixin_scan=True
269      )
270  
271  
272  def _metaclass_cls_hook(ctx: ClassDefContext) -> None:
273      util.set_is_base(ctx.cls.info)
274  
275  
276  def _base_cls_hook(ctx: ClassDefContext) -> None:
277      _add_globals(ctx)
278      decl_class.scan_declarative_assignments_and_apply_types(ctx.cls, ctx.api)

281  def _queryable_getattr_hook(ctx: AttributeContext) -> Type:
282      # how do I....tell it it has no attribute of a certain name?
283      # can't find any Type that seems to match that
284      return ctx.default_attr_type

287  def _add_globals(ctx: Union[ClassDefContext, DynamicClassDefContext]) -> None:
288      """Add __sa_DeclarativeMeta and __sa_Mapped symbol to the global space 
289      for all class defs 
290   
291      """
292  
293      util.add_global(ctx, "sqlalchemy.orm", "Mapped", "__sa_Mapped")

296  def _set_declarative_metaclass(
297      api: SemanticAnalyzerPluginInterface, target_cls: ClassDef
298  ) -> None:
299      info = target_cls.info
300      sym = api.lookup_fully_qualified_or_none(
301          "sqlalchemy.orm.decl_api.DeclarativeMeta"
302      )
303      assert sym is not None and isinstance(sym.node, TypeInfo)
304      info.declared_metaclass = info.metaclass_type = Instance(sym.node, [])
