27   class HasFinalizerTracksInstances(object):
28       EXTANT_INSTANCES = set()
29       def __init__(self, msg):
30           self.msg = sys.intern(msg)
31           self.EXTANT_INSTANCES.add(id(self))
32       def __del__(self):
33           self.EXTANT_INSTANCES.remove(id(self))
34       def __repr__(self):
35           return "<HasFinalizerTracksInstances at 0x%x %r>" % (
36               id(self), self.msg
37           )
38       @classmethod
39       def reset(cls):
40           cls.EXTANT_INSTANCES.clear()

53       def test_arg_refs(self):
54           args = ('a', 'b', 'c')
55           refcount_before = sys.getrefcount(args)
56           # pylint:disable=unnecessary-lambda
57           g = greenlet.greenlet(
58               lambda *args: greenlet.getcurrent().parent.switch(*args))
59           for _ in range(100):
60               g.switch(*args)
61           self.assertEqual(sys.getrefcount(args), refcount_before)

63       def test_kwarg_refs(self):
64           kwargs = {}
65           self.assertEqual(sys.getrefcount(kwargs), 2 if not PY314 else 1)
66           # pylint:disable=unnecessary-lambda
67           g = greenlet.greenlet(
68               lambda **gkwargs: greenlet.getcurrent().parent.switch(**gkwargs))
69           for _ in range(100):
70               g.switch(**kwargs)
71           # Python 3.14 elides reference counting operations
72           # in some cases. See https://github.com/python/cpython/pull/130708
73           self.assertEqual(sys.getrefcount(kwargs), 2 if not PY314 else 1)

76       @staticmethod
77       def __recycle_threads():
78           # By introducing a thread that does sleep we allow other threads,
79           # that have triggered their __block condition, but did not have a
80           # chance to deallocate their thread state yet, to finally do so.
81           # The way it works is by requiring a GIL switch (different thread),
82           # which does a GIL release (sleep), which might do a GIL switch
83           # to finished threads and allow them to clean up.
84           def worker():
85               time.sleep(0.001)
86           t = threading.Thread(target=worker)
87           t.start()
88           time.sleep(0.001)
89           t.join(10)

91       def test_threaded_leak(self):
92           gg = []
93           def worker():
94               # only main greenlet present
95               gg.append(weakref.ref(greenlet.getcurrent()))
96           for _ in range(2):
97               t = threading.Thread(target=worker)
98               t.start()
99               t.join(10)
100              del t
101          greenlet.getcurrent() # update ts_current
102          self.__recycle_threads()
103          greenlet.getcurrent() # update ts_current
104          gc.collect()
105          greenlet.getcurrent() # update ts_current
106          for g in gg:
107              self.assertIsNone(g())

110          gg = []
111          def worker():
112              # main and additional *finished* greenlets
113              ll = greenlet.getcurrent().ll = []
114              def additional():
115                  ll.append(greenlet.getcurrent())
116              for _ in range(2):
117                  greenlet.greenlet(additional).switch()

126          greenlet.getcurrent() # update ts_current
127          gc.collect()
128          greenlet.getcurrent() # update ts_current
129          for g in gg:
130              self.assertIsNone(g())

133          used = greenlet._greenlet.get_clocks_used_doing_optional_cleanup()
134          self.assertGreaterEqual(used, 0)
135          # we don't lose the value
136          greenlet._greenlet.enable_optional_cleanup(True)
137          used2 = greenlet._greenlet.get_clocks_used_doing_optional_cleanup()
138          self.assertEqual(used, used2)
139          self.assertGreater(greenlet._greenlet.CLOCKS_PER_SEC, 1)

141      def _check_issue251(self,
142                          manually_collect_background=True,
143                          explicit_reference_to_switch=False):
144          # See https://github.com/python-greenlet/greenlet/issues/251
145          # Killing a greenlet (probably not the main one)
146          # in one thread from another thread would
147          # result in leaking a list (the ts_delkey list).
148          # We no longer use lists to hold that stuff, though.
149  
150          # For the test to be valid, even empty lists have to be tracked by the
151          # GC
152  
153          assert gc.is_tracked([])
154          HasFinalizerTracksInstances.reset()
155          greenlet.getcurrent()
156          greenlets_before = self.count_objects(greenlet.greenlet, exact_kind=False)
157  
158          background_glet_running = threading.Event()
159          background_glet_killed = threading.Event()
160          background_greenlets = []
161  
162          # XXX: Switching this to a greenlet subclass that overrides
163          # run results in all callers failing the leaktest; that
164          # greenlet instance is leaked. There's a bound method for
165          # run() living on the stack of the greenlet in g_initialstub,
166          # and since we don't manually switch back to the background
167          # greenlet to let it "fall off the end" and exit the
168          # g_initialstub function, it never gets cleaned up. Making the
169          # garbage collector aware of this bound method (making it an
170          # attribute of the greenlet structure and traversing into it)
171          # doesn't help, for some reason.
172          def background_greenlet():
173              # Throw control back to the main greenlet.
174              jd = HasFinalizerTracksInstances("DELETING STACK OBJECT")
175              greenlet._greenlet.set_thread_local(
176                  'test_leaks_key',
177                  HasFinalizerTracksInstances("DELETING THREAD STATE"))
178              # Explicitly keeping 'switch' in a local variable
179              # breaks this test in all versions
180              if explicit_reference_to_switch:
181                  s = greenlet.getcurrent().parent.switch
182                  s([jd])
183              else:
184                  greenlet.getcurrent().parent.switch([jd])
185  
186          bg_main_wrefs = []
187  
188          def background_thread():
189              glet = greenlet.greenlet(background_greenlet)
190              bg_main_wrefs.append(weakref.ref(glet.parent))
191  
192              background_greenlets.append(glet)
193              glet.switch() # Be sure it's active.
194              # Control is ours again.
195              del glet # Delete one reference from the thread it runs in.
196              background_glet_running.set()
197              background_glet_killed.wait(10)
198  
199              # To trigger the background collection of the dead
200              # greenlet, thus clearing out the contents of the list, we
201              # need to run some APIs. See issue 252.
202              if manually_collect_background:
203                  greenlet.getcurrent()
204  
205  
206          t = threading.Thread(target=background_thread)
207          t.start()
208          background_glet_running.wait(10)
209          greenlet.getcurrent()
210          lists_before = self.count_objects(list, exact_kind=True)
211  
212          assert len(background_greenlets) == 1
213          self.assertFalse(background_greenlets[0].dead)
214          # Delete the last reference to the background greenlet
215          # from a different thread. This puts it in the background thread's
216          # ts_delkey list.
217          del background_greenlets[:]
218          background_glet_killed.set()
219  
220          # Now wait for the background thread to die.
221          t.join(10)
222          del t
223          # As part of the fix for 252, we need to cycle the ceval.c
224          # interpreter loop to be sure it has had a chance to process
225          # the pending call.
226          self.wait_for_pending_cleanups()
227  
228          lists_after = self.count_objects(list, exact_kind=True)
229          greenlets_after = self.count_objects(greenlet.greenlet, exact_kind=False)
230  
231          # On 2.7, we observe that lists_after is smaller than
232          # lists_before. No idea what lists got cleaned up. All the
233          # Python 3 versions match exactly.
234          self.assertLessEqual(lists_after, lists_before)
235          # On versions after 3.6, we've successfully cleaned up the
236          # greenlet references thanks to the internal "vectorcall"
237          # protocol; prior to that, there is a reference path through
238          # the ``greenlet.switch`` method still on the stack that we
239          # can't reach to clean up. The C code goes through terrific
240          # lengths to clean that up.
241          if not explicit_reference_to_switch \
242             and greenlet._greenlet.get_clocks_used_doing_optional_cleanup() is not None:
243              # If cleanup was disabled, though, we may not find it.
244              self.assertEqual(greenlets_after, greenlets_before)
245              if manually_collect_background:
246                  # TODO: Figure out how to make this work!
247                  # The one on the stack is still leaking somehow
248                  # in the non-manually-collect state.
249                  self.assertEqual(HasFinalizerTracksInstances.EXTANT_INSTANCES, set())
250          else:
251              # The explicit reference prevents us from collecting it
252              # and it isn't always found by the GC either for some
253              # reason. The entire frame is leaked somehow, on some
254              # platforms (e.g., MacPorts builds of Python (all
255              # versions!)), but not on other platforms (the linux and
256              # windows builds on GitHub actions and Appveyor). So we'd
257              # like to write a test that proves that the main greenlet
258              # sticks around, and we can on my machine (macOS 11.6,
259              # MacPorts builds of everything) but we can't write that
260              # same test on other platforms. However, hopefully iteration
261              # done by leakcheck will find it.
262              pass
263  
264          if greenlet._greenlet.get_clocks_used_doing_optional_cleanup() is not None:
265              self.assertClocksUsed()

326      @ignores_leakcheck
327      # Because we're just trying to track raw memory, not objects, and running
328      # the leakcheck makes an already slow test slower.
329      def test_untracked_memory_doesnt_increase(self):
330          # See https://github.com/gevent/gevent/issues/1924
331          # and https://github.com/python-greenlet/greenlet/issues/328
332          self._only_test_some_versions()
333          def f():
334              return 1
335  
336          ITER = 10000
337          def run_it():
338              for _ in range(ITER):
339                  greenlet.greenlet(f).switch()
340  
341          # Establish baseline
342          for _ in range(3):
343              run_it()
344  
345          # uss: (Linux, macOS, Windows): aka "Unique Set Size", this is
346          # the memory which is unique to a process and which would be
347          # freed if the process was terminated right now.
348          uss_before = self.get_process_uss()
349  
350          for count in range(self.UNTRACK_ATTEMPTS):
351              uss_before = max(uss_before, self.get_process_uss())
352              run_it()
353  
354              uss_after = self.get_process_uss()
355              if uss_after <= uss_before and count > 1:
356                  break
357  
358          self.assertLessEqual(uss_after, uss_before)

360      def _check_untracked_memory_thread(self, deallocate_in_thread=True):
361          self._only_test_some_versions()
362          # Like the above test, but what if there are a bunch of
363          # unfinished greenlets in a thread that dies?
364          # Does it matter if we deallocate in the thread or not?
365  
366          # First, make sure we can get useful measurements. This will
367          # be skipped if not.
368          self.get_process_uss()
369  
370          EXIT_COUNT = [0]
371  
372          def f():
373              try:
374                  greenlet.getcurrent().parent.switch()
375              except greenlet.GreenletExit:
376                  EXIT_COUNT[0] += 1
377                  raise
378              return 1
379  
380          ITER = 10000
381          def run_it():
382              glets = []
383              for _ in range(ITER):
384                  # Greenlet starts, switches back to us.
385                  # We keep a strong reference to the greenlet though so it doesn't
386                  # get a GreenletExit exception.
387                  g = greenlet.greenlet(f)
388                  glets.append(g)
389                  g.switch()
390  
391              return glets
392  
393          test = self
394  
395          class ThreadFunc:
396              uss_before = uss_after = 0
397              glets = ()
398              ITER = 2
399              def __call__(self):
400                  self.uss_before = test.get_process_uss()
401  
402                  for _ in range(self.ITER):
403                      self.glets += tuple(run_it())
404  
405                  for g in self.glets:
406                      test.assertIn('suspended active', str(g))
407                  # Drop them.
408                  if deallocate_in_thread:
409                      self.glets = ()
410                  self.uss_after = test.get_process_uss()
411  
412          # Establish baseline
413          uss_before = uss_after = None
414          for count in range(self.UNTRACK_ATTEMPTS):
415              EXIT_COUNT[0] = 0
416              thread_func = ThreadFunc()
417              t = threading.Thread(target=thread_func)
418              t.start()
419              t.join(30)
420              self.assertFalse(t.is_alive())
421  
422              if uss_before is None:
423                  uss_before = thread_func.uss_before
424  
425              uss_before = max(uss_before, thread_func.uss_before)
426              if deallocate_in_thread:
427                  self.assertEqual(thread_func.glets, ())
428                  self.assertEqual(EXIT_COUNT[0], ITER * thread_func.ITER)
429  
430              del thread_func # Deallocate the greenlets; but this won't raise into them
431              del t
432              if not deallocate_in_thread:
433                  self.assertEqual(EXIT_COUNT[0], 0)
434              if deallocate_in_thread:
435                  self.wait_for_pending_cleanups()
436  
437              uss_after = self.get_process_uss()
438              # See if we achieve a non-growth state at some point. Break when we do.
439              if uss_after <= uss_before and count > 1:
440                  break
441  
442          self.wait_for_pending_cleanups()
443          uss_after = self.get_process_uss()
444          # On Windows, USS can fluctuate by tens of KB between measurements
445          # due to working set trimming, page table updates, etc. Allow a
446          # small tolerance so OS-level noise doesn't cause false failures.
447          # Real leaks produce MBs of growth (each iteration creates 20k
448          # greenlets; the greenlet Python object alone, not counting the C++
449          # state it points to, is 56 bytes so 20k greenlets alone is
450          # slightly over a megabyte; the C++ UserGreenlet clocks in at 208
451          # bytes, which adds another 4MB), so 512 KB is well below the detection
452          # threshold for genuine issues.
453          #
454          # 2025-06-16: We have now observed the same problem on both
455          # ubuntu-latest 3.14t (2948K "leak") and macos-latest 3.15t (192K
456          # "leak"). This was after the merge of PR #511, but that change
457          # should be unrelated. In particular, the ubuntu failure was after
458          # 30 loops and the macos failure was after only 3 --- both much
459          # smaller than UNTRACK_ATTEMPTS=100! We break out of the loop if we
460          # see a USS at the end of the loop that's smaller than at the start
461          # of the loop, and then we wait for pending cleanups and get the
462          # USS again. So this means that we think we have success, break out
463          # of the loop, and the pending cleanups cause our USS to grow -
464          # perhaps by touching pages that had been paged out? An immediately
465          # prior attempt of both platforms had been successful.
466          #
467          # At any rate, we need a larger tolerance than 512K, and we
468          # need it on all platforms, not just Windows.
469          tolerance = 3 * 1024 * 1024
470          self.assertLessEqual(uss_after, uss_before + tolerance,
471                               "after attempts %d" % (count,))
