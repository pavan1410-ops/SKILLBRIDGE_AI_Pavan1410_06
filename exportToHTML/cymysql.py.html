47   class _cymysqlBIT(BIT):
48       def result_processor(
49           self, dialect: Dialect, coltype: object
50       ) -> Optional[_ResultProcessorType[Any]]:
51           """Convert MySQL's 64 bit, variable length binary string to a long."""
52   
53           def process(value: Optional[Iterable[int]]) -> Optional[int]:
54               if value is not None:
55                   v = 0
56                   for i in iter(value):
57                       v = v << 8 | i
58                   return v
59               return value
60   
61           return process

64   class MySQLDialect_cymysql(MySQLDialect_mysqldb):
65       driver = "cymysql"
66       supports_statement_cache = True
67   
68       description_encoding = None
69       supports_sane_rowcount = True
70       supports_sane_multi_rowcount = False
71       supports_unicode_statements = True
72   
73       colspecs = util.update_copy(MySQLDialect.colspecs, {BIT: _cymysqlBIT})
74   
75       @classmethod
76       def import_dbapi(cls) -> DBAPIModule:
77           return __import__("cymysql")
78   
79       def _detect_charset(self, connection: Connection) -> str:
80           return connection.connection.charset  # type: ignore[no-any-return]
81   
82       def _extract_error_code(self, exception: DBAPIModule.Error) -> int:
83           return exception.errno  # type: ignore[no-any-return]
84   
85       def is_disconnect(
86           self,
87           e: DBAPIModule.Error,
88           connection: Optional[Union[PoolProxiedConnection, DBAPIConnection]],
89           cursor: Optional[DBAPICursor],
90       ) -> bool:
91           if isinstance(e, self.loaded_dbapi.OperationalError):
92               return self._extract_error_code(e) in (
93                   2006,
94                   2013,
95                   2014,
96                   2045,
97                   2055,
98               )
99           elif isinstance(e, self.loaded_dbapi.InterfaceError):
100              # if underlying connection is closed,
101              # this is the error you get
102              return True
103          else:
104              return False
