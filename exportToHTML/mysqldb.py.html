16   Driver Status 
17   ------------- 
18    
19   The mysqlclient DBAPI is a maintained fork of the 
20   `MySQL-Python <https://sourceforge.net/projects/mysql-python>`_ DBAPI 
21   that is no longer maintained.  `mysqlclient`_ supports Python 2 and Python 3 
22   and is very stable. 
23    
24   .. _mysqlclient: https://github.com/PyMySQL/mysqlclient-python 
25    
26   .. _mysqldb_unicode: 
27    
28   Unicode 
29   ------- 
30    
31   Please see :ref:`mysql_unicode` for current recommendations on unicode 
32   handling. 
33    
34   .. _mysqldb_ssl: 
35    
36   SSL Connections 
37   ---------------- 
38    
39   The mysqlclient and PyMySQL DBAPIs accept an additional dictionary under the 
40   key "ssl", which may be specified using the 
41   :paramref:`_sa.create_engine.connect_args` dictionary:: 
42    
43       engine = create_engine( 
44           "mysql+mysqldb://scott:tiger@192.168.0.134/test", 
45           connect_args={ 
46               "ssl": { 
47                   "ca": "/home/gord/client-ssl/ca.pem", 
48                   "cert": "/home/gord/client-ssl/client-cert.pem", 
49                   "key": "/home/gord/client-ssl/client-key.pem", 
50               } 
51           }, 
52       ) 
53    
54   For convenience, the following keys may also be specified inline within the URL 
55   where they will be interpreted into the "ssl" dictionary automatically: 
56   "ssl_ca", "ssl_cert", "ssl_key", "ssl_capath", "ssl_cipher", 
57   "ssl_check_hostname". An example is as follows:: 
58    
59       connection_uri = ( 
60           "mysql+mysqldb://scott:tiger@192.168.0.134/test" 
61           "?ssl_ca=/home/gord/client-ssl/ca.pem" 
62           "&ssl_cert=/home/gord/client-ssl/client-cert.pem" 
63           "&ssl_key=/home/gord/client-ssl/client-key.pem" 
64       ) 
65    
66   .. seealso:: 
67    
68       :ref:`pymysql_ssl` in the PyMySQL dialect 
69    
70    
71   Using MySQLdb with Google Cloud SQL 
72   ----------------------------------- 
73    
74   Google Cloud SQL now recommends use of the MySQLdb dialect.  Connect 
75   using a URL like the following: 
76    
77   .. sourcecode:: text 
78    
79       mysql+mysqldb://root@/<dbname>?unix_socket=/cloudsql/<projectid>:<instancename>

142      def __init__(self, **kwargs: Any):
143          super().__init__(**kwargs)
144          self._mysql_dbapi_version = (
145              self._parse_dbapi_version(self.dbapi.__version__)
146              if self.dbapi is not None and hasattr(self.dbapi, "__version__")
147              else (0, 0, 0)

150      def _parse_dbapi_version(self, version: str) -> Tuple[int, ...]:
151          m = re.match(r"(\d+)\.(\d+)(?:\.(\d+))?", version)
152          if m:
153              return tuple(int(x) for x in m.group(1, 2, 3) if x is not None)
154          else:
155              return (0, 0, 0)

157      @util.langhelpers.memoized_property

170      def on_connect(self) -> Callable[[DBAPIConnection], None]:
171          super_ = super().on_connect()
172  
173          def on_connect(conn: DBAPIConnection) -> None:
174              if super_ is not None:
175                  super_(conn)
176  
177              charset_name = conn.character_set_name()
178  
179              if charset_name is not None:
180                  cursor = conn.cursor()
181                  cursor.execute("SET NAMES %s" % charset_name)
182                  cursor.close()
183  
184          return on_connect

201      def create_connect_args(
202          self, url: URL, _translate_args: Optional[Dict[str, Any]] = None
203      ) -> ConnectArgsType:
204          if _translate_args is None:
205              _translate_args = dict(
206                  database="db", username="user", password="passwd"
207              )
208  
209          opts = url.translate_connect_args(**_translate_args)
210          opts.update(url.query)
211  
212          util.coerce_kw_type(opts, "compress", bool)
213          util.coerce_kw_type(opts, "connect_timeout", int)
214          util.coerce_kw_type(opts, "read_timeout", int)
215          util.coerce_kw_type(opts, "write_timeout", int)
216          util.coerce_kw_type(opts, "client_flag", int)
217          util.coerce_kw_type(opts, "local_infile", bool)
218          # Note: using either of the below will cause all strings to be
219          # returned as Unicode, both in raw SQL operations and with column
220          # types like String and MSString.
221          util.coerce_kw_type(opts, "use_unicode", bool)
222          util.coerce_kw_type(opts, "charset", str)
223  
224          # Rich values 'cursorclass' and 'conv' are not supported via
225          # query string.
226  
227          ssl = {}
228          keys = [
229              ("ssl_ca", str),
230              ("ssl_key", str),
231              ("ssl_cert", str),
232              ("ssl_capath", str),
233              ("ssl_cipher", str),
234              ("ssl_check_hostname", bool),
235          ]
236          for key, kw_type in keys:
237              if key in opts:
238                  ssl[key[4:]] = opts[key]
239                  util.coerce_kw_type(ssl, key[4:], kw_type)
240                  del opts[key]
241          if ssl:
242              opts["ssl"] = ssl
243  
244          # FOUND_ROWS must be set in CLIENT_FLAGS to enable
245          # supports_sane_rowcount.
246          client_flag = opts.get("client_flag", 0)
247  
248          client_flag_found_rows = self._found_rows_client_flag()
249          if client_flag_found_rows is not None:
250              client_flag |= client_flag_found_rows
251              opts["client_flag"] = client_flag
252          return [], opts
253  
254      def _found_rows_client_flag(self) -> Optional[int]:
255          if self.dbapi is not None:
256              try:
257                  CLIENT_FLAGS = __import__(
258                      self.dbapi.__name__ + ".constants.CLIENT"
259                  ).constants.CLIENT
260              except (AttributeError, ImportError):
261                  return None
262              else:
263                  return CLIENT_FLAGS.FOUND_ROWS  # type: ignore
264          else:
265              return None

270      def _detect_charset(self, connection: Connection) -> str:
271          """Sniff out the character set in use for connection results."""
272  
273          try:
274              # note: the SQL here would be
275              # "SHOW VARIABLES LIKE 'character_set%%'"
276  
277              cset_name: Callable[[], str] = (
278                  connection.connection.character_set_name
279              )
280          except AttributeError:
281              util.warn(
282                  "No 'character_set_name' can be detected with "
283                  "this MySQL-Python version; "
284                  "please upgrade to a recent version of MySQL-Python.  "
285                  "Assuming latin1."
286              )
287              return "latin1"
288          else:
289              return cset_name()

302      def detect_autocommit_setting(self, dbapi_conn: DBAPIConnection) -> bool:
303          return dbapi_conn.get_autocommit()  # type: ignore[no-any-return]

305      def set_isolation_level(
306          self, dbapi_connection: DBAPIConnection, level: IsolationLevel
307      ) -> None:
308          if level == "AUTOCOMMIT":
309              dbapi_connection.autocommit(True)
310          else:
311              dbapi_connection.autocommit(False)
312              super().set_isolation_level(dbapi_connection, level)
