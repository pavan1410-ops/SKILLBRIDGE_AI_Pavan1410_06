1    from flask.signals import Namespace
2    
3    _signals = Namespace()
4    
5    #: Sent when a user is logged in. In addition to the app (which is the
6    #: sender), it is passed `user`, which is the user being logged in.
7    user_logged_in = _signals.signal("logged-in")
8    
9    #: Sent when a user is logged out. In addition to the app (which is the
10   #: sender), it is passed `user`, which is the user being logged out.
11   user_logged_out = _signals.signal("logged-out")
12   
13   #: Sent when the user is loaded from the cookie. In addition to the app (which
14   #: is the sender), it is passed `user`, which is the user being reloaded.
15   user_loaded_from_cookie = _signals.signal("loaded-from-cookie")
16   
17   #: Sent when the user is loaded from the header. In addition to the app (which
18   #: is the #: sender), it is passed `user`, which is the user being reloaded.
19   _user_loaded_from_header = _signals.signal("loaded-from-header")
20   
21   #: Sent when the user is loaded from the request. In addition to the app (which
22   #: is the #: sender), it is passed `user`, which is the user being reloaded.
23   user_loaded_from_request = _signals.signal("loaded-from-request")
24   
25   #: Sent when a user's login is confirmed, marking it as fresh. (It is not
26   #: called for a normal login.)
27   #: It receives no additional arguments besides the app.
28   user_login_confirmed = _signals.signal("login-confirmed")
29   
30   #: Sent when the `unauthorized` method is called on a `LoginManager`. It
31   #: receives no additional arguments besides the app.
32   user_unauthorized = _signals.signal("unauthorized")
33   
34   #: Sent when the `needs_refresh` method is called on a `LoginManager`. It
35   #: receives no additional arguments besides the app.
36   user_needs_refresh = _signals.signal("needs-refresh")
37   
38   #: Sent whenever the user is accessed/loaded
39   #: receives no additional arguments besides the app.
40   user_accessed = _signals.signal("accessed")
41   
42   #: Sent whenever session protection takes effect, and a session is either
43   #: marked non-fresh or deleted. It receives no additional arguments besides
44   #: the app.
45   session_protected = _signals.signal("session-protected")
46   
47
