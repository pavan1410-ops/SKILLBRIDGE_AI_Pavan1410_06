15    
16   Driver Status 
17   ------------- 
18    
19   MariaDB Connector/Python enables Python programs to access MariaDB and MySQL 
20   databases using an API which is compliant with the Python DB API 2.0 (PEP-249). 
21   It is written in C and uses MariaDB Connector/C client library for client server 
22   communication. 
23    
24   Note that the default driver for a ``mariadb://`` connection URI continues to 
25   be ``mysqldb``. ``mariadb+mariadbconnector://`` is required to use this driver. 
26    
27   .. mariadb: https://github.com/mariadb-corporation/mariadb-connector-python

66   class _MariaDBUUID(sqltypes.UUID[sqltypes._UUID_RETURN]):
67       # work around JIRA issue
68       # https://jira.mariadb.org/browse/CONPY-270.  When that issue is fixed,
69       # this type can be removed.
70       def result_processor(
71           self, dialect: Dialect, coltype: object
72       ) -> Optional[_ResultProcessorType[Any]]:
73           if self.as_uuid:
74   
75               def process(value: Any) -> Any:
76                   if value is not None:
77                       if hasattr(value, "decode"):
78                           value = value.decode("ascii")
79                       value = _python_UUID(value)
80                   return value
81   
82               return process
83           else:
84   
85               def process(value: Any) -> Any:
86                   if value is not None:
87                       if hasattr(value, "decode"):
88                           value = value.decode("ascii")
89                       value = str(_python_UUID(value))
90                   return value
91   
92               return process

104      def post_exec(self) -> None:
105          super().post_exec()
106  
107          self._rowcount = self.cursor.rowcount
108  
109          if TYPE_CHECKING:
110              assert isinstance(self.compiled, SQLCompiler)
111          if self.isinsert and self.compiled.postfetch_lastrowid:

115          if TYPE_CHECKING:

124  class MySQLDialect_mariadbconnector(MySQLDialect):
125      driver = "mariadbconnector"
126      supports_statement_cache = True

128      # set this to True at the module level to prevent the driver from running
129      # against a backend that server detects as MySQL. currently this appears to
130      # be unnecessary as MariaDB client libraries have always worked against
131      # MySQL databases.   However, if this changes at some point, this can be
132      # adjusted, but PLEASE ADD A TEST in test/dialect/mysql/test_dialect.py if
133      # this change is made at some point to ensure the correct exception
134      # is raised at the correct point when running the driver against
135      # a MySQL backend.
136      # is_mariadb = True

138      supports_unicode_statements = True

154      @util.memoized_property
155      def _dbapi_version(self) -> Tuple[int, ...]:
156          if self.dbapi and hasattr(self.dbapi, "__version__"):
157              return tuple(
158                  [
159                      int(x)
160                      for x in re.findall(
161                          r"(\d+)(?:[-\.]?|$)", self.dbapi.__version__
162                      )
163                  ]
164              )
165          else:
166              return (99, 99, 99)

168      def __init__(self, **kwargs: Any) -> None:
169          super().__init__(**kwargs)
170          self.paramstyle = "qmark"
171          if self.dbapi is not None:
172              if self._dbapi_version < mariadb_cpy_minimum_version:
173                  raise NotImplementedError(
174                      "The minimum required version for MariaDB "
175                      "Connector/Python is %s"
176                      % ".".join(str(x) for x in mariadb_cpy_minimum_version)
177                  )

183      def is_disconnect(
184          self,
185          e: DBAPIModule.Error,
186          connection: Optional[Union[PoolProxiedConnection, DBAPIConnection]],
187          cursor: Optional[DBAPICursor],
188      ) -> bool:
189          if super().is_disconnect(e, connection, cursor):
190              return True
191          elif isinstance(e, self.loaded_dbapi.Error):
192              str_e = str(e).lower()
193              return "not connected" in str_e or "isn't valid" in str_e
194          else:
195              return False

197      def create_connect_args(self, url: URL) -> ConnectArgsType:
198          opts = url.translate_connect_args()
199          opts.update(url.query)
200  
201          int_params = [
202              "connect_timeout",
203              "read_timeout",
204              "write_timeout",
205              "client_flag",
206              "port",
207              "pool_size",
208          ]
209          bool_params = [
210              "local_infile",
211              "ssl_verify_cert",
212              "ssl",
213              "pool_reset_connection",
214              "compress",
215          ]
216  
217          for key in int_params:
218              util.coerce_kw_type(opts, key, int)
219          for key in bool_params:
220              util.coerce_kw_type(opts, key, bool)
221  
222          # FOUND_ROWS must be set in CLIENT_FLAGS to enable
223          # supports_sane_rowcount.
224          client_flag = opts.get("client_flag", 0)
225          if self.dbapi is not None:
226              try:
227                  CLIENT_FLAGS = __import__(
228                      self.dbapi.__name__ + ".constants.CLIENT"
229                  ).constants.CLIENT
230                  client_flag |= CLIENT_FLAGS.FOUND_ROWS
231              except (AttributeError, ImportError):
232                  self.supports_sane_rowcount = False
233              opts["client_flag"] = client_flag
234          return [], opts

269      def do_begin_twophase(self, connection: Connection, xid: Any) -> None:
270          connection.execute(
271              sql.text("XA BEGIN :xid").bindparams(
272                  sql.bindparam("xid", xid, literal_execute=True)
273              )
274          )
275  
276      def do_prepare_twophase(self, connection: Connection, xid: Any) -> None:
277          connection.execute(
278              sql.text("XA END :xid").bindparams(
279                  sql.bindparam("xid", xid, literal_execute=True)
280              )
281          )
282          connection.execute(
283              sql.text("XA PREPARE :xid").bindparams(
284                  sql.bindparam("xid", xid, literal_execute=True)
285              )
286          )
287  
288      def do_rollback_twophase(
289          self,
290          connection: Connection,
291          xid: Any,
292          is_prepared: bool = True,
293          recover: bool = False,
294      ) -> None:
295          if not is_prepared:
296              connection.execute(
297                  sql.text("XA END :xid").bindparams(
298                      sql.bindparam("xid", xid, literal_execute=True)
299                  )
300              )
301          connection.execute(
302              sql.text("XA ROLLBACK :xid").bindparams(
303                  sql.bindparam("xid", xid, literal_execute=True)
304              )
305          )
306  
307      def do_commit_twophase(
308          self,
309          connection: Connection,
310          xid: Any,
311          is_prepared: bool = True,
312          recover: bool = False,
313      ) -> None:
314          if not is_prepared:
315              self.do_prepare_twophase(connection, xid)
316          connection.execute(
317              sql.text("XA COMMIT :xid").bindparams(
318                  sql.bindparam("xid", xid, literal_execute=True)
319              )
320          )
