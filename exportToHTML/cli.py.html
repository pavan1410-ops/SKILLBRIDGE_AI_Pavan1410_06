8    def add_models_to_shell() -> dict[str, t.Any]:
9        """Registered with :meth:`~flask.Flask.shell_context_processor` if 
10       ``add_models_to_shell`` is enabled. Adds the ``db`` instance and all model classes 
11       to ``flask shell``. 
12       """

14       out = {m.class_.__name__: m.class_ for m in db.Model._sa_registry.mappers}
15       out["db"] = db
