1    """ 
2    Testing initialstub throwing an already started exception. 
3    """
4    
5    import greenlet
6    
7    a = None
8    b = None
9    c = None
10   main = greenlet.getcurrent()
11   
12   # If we switch into a dead greenlet,
13   # we go looking for its parents.
14   # if a parent is not yet started, we start it.
15   
16   results = []
17   
18   def a_run(*args):
19       #results.append('A')
20       results.append(('Begin A', args))
21   
22   
23   def c_run():
24       results.append('Begin C')
25       b.switch('From C')
26       results.append('C done')
27   
28   class A(greenlet.greenlet): pass
29   
30   class B(greenlet.greenlet):
31       doing_it = False
32       def __getattribute__(self, name):
33           if name == 'run' and not self.doing_it:
34               assert greenlet.getcurrent() is c
35               self.doing_it = True
36               results.append('Switch to b from B.__getattribute__ in '
37                              + type(greenlet.getcurrent()).__name__)
38               b.switch()
39               results.append('B.__getattribute__ back from main in '
40                              + type(greenlet.getcurrent()).__name__)
41           if name == 'run':
42               name = '_B_run'
43           return object.__getattribute__(self, name)
44   
45       def _B_run(self, *arg):
46           results.append(('Begin B', arg))
47           results.append('_B_run switching to main')
48           main.switch('From B')
49   
50   class C(greenlet.greenlet):
51       pass
52   a = A(a_run)
53   b = B(parent=a)
54   c = C(c_run, b)
55   
56   # Start a child; while running, it will start B,
57   # but starting B will ALSO start B.
58   result = c.switch()
59   results.append(('main from c', result))
60   
61   # Switch back to C, which was in the middle of switching
62   # already. This will throw the ``GreenletStartedWhileInPython``
63   # exception, which results in parent A getting started (B is finished)
64   c.switch()
65   
66   results.append(('A dead?', a.dead, 'B dead?', b.dead, 'C dead?', c.dead))
67   
68   # A and B should both be dead now.
69   assert a.dead
70   assert b.dead
71   assert not c.dead
72   
73   result = c.switch()
74   results.append(('main from c.2', result))
75   # Now C is dead
76   assert c.dead
77   
78   print("RESULTS:", results)
