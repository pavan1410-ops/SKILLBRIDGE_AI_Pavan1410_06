269      class SwitchingArgs
270      {
271      private:
272          G_NO_ASSIGNMENT_OF_CLS(SwitchingArgs);
273          // If args and kwargs are both false (NULL), this is a *throw*, not a
274          // switch. PyErr_... must have been called already.
275          OwnedObject _args;
276          OwnedObject _kwargs;
277      public:
278  
279          SwitchingArgs()
280          {}
281  
282          SwitchingArgs(const OwnedObject& args, const OwnedObject& kwargs)
283              : _args(args),
284                _kwargs(kwargs)
285          {}
286  
287          SwitchingArgs(const SwitchingArgs& other)
288              : _args(other._args),
289                _kwargs(other._kwargs)
290          {}
291  
292          const OwnedObject& args()
293          {
294              return this->_args;
295          }
296  
297          const OwnedObject& kwargs()
298          {
299              return this->_kwargs;
300          }
301  
302          /** 
303           * Moves ownership from the argument to this object. 
304           */
305          SwitchingArgs& operator<<=(SwitchingArgs& other)
306          {
307              if (this != &other) {
308                  this->_args = other._args;
309                  this->_kwargs = other._kwargs;
310                  other.CLEAR();
311              }
312              return *this;
313          }
314  
315          /** 
316           * Acquires ownership of the argument (consumes the reference). 
317           */
318          SwitchingArgs& operator<<=(PyObject* args)
319          {
320              this->_args = OwnedObject::consuming(args);
321              this->_kwargs.CLEAR();
322              return *this;
323          }
324  
325          /** 
326           * Acquires ownership of the argument. 
327           * 
328           * Sets the args to be the given value; clears the kwargs. 
329           */
330          SwitchingArgs& operator<<=(OwnedObject& args)
331          {
332              assert(&args != &this->_args);
333              this->_args = args;
334              this->_kwargs.CLEAR();
335              args.CLEAR();
336  
337              return *this;
338          }
339  
340          explicit operator bool() const noexcept
341          {
342              return this->_args || this->_kwargs;
343          }
344  
345          inline void CLEAR()
346          {
347              this->_args.CLEAR();
348              this->_kwargs.CLEAR();
349          }
350  
351          const std::string as_str() const noexcept
352          {
353              return PyUnicode_AsUTF8(
354                  OwnedObject::consuming(
355                      PyUnicode_FromFormat(
356                          "SwitchingArgs(args=%R, kwargs=%R)",
357                          this->_args.borrow(),
358                          this->_kwargs.borrow()
359                      )
360                  ).borrow()
361              );
362          }
363      };

623   
624             This temporarily sets the global variable 
625             ``switching_thread_state`` to this greenlet; as soon as the 
626             call to ``slp_switch`` completes, this is reset to NULL. 
627             Consequently, this depends on the GIL.

673      inline void CallTraceFunction(const OwnedObject& tracefunc,
674                                    const greenlet::refs::ImmortalEventName& event,
675                                    const BorrowedGreenlet& origin,
676                                    const BorrowedGreenlet& target)
677      {
678          // TODO: This calls tracefunc(event, (origin, target)). Add a shortcut
679          // function for that that's specialized to avoid the Py_BuildValue
680          // string parsing, or start with just using "ON" format with PyTuple_Pack(2,
681          // origin, target). That seems like what the N format is meant
682          // for.
683          // XXX: Why does event not automatically cast back to a PyObject?
684          // It tries to call the "deleted constructor ImmortalEventName
685          // const" instead.
686          assert(tracefunc);
687          assert(event);
688          assert(origin);
689          assert(target);
690          greenlet::refs::NewReference retval(
691              PyObject_CallFunction(
692                  tracefunc.borrow(),
693                  "O(OO)",
694                  event.borrow(),
695                  origin.borrow(),
696                  target.borrow()
697              ));
698          if (!retval) {
699              throw PyErrOccurred::from_current();
700          }
701      }
702  };

826      {
827      private:
828          int was_enabled = 0;

832          {
833              PyGC_Disable();
834          }

850      {
851          if (results
852              && PyTuple_Check(results.borrow())
853              && PyTuple_GET_SIZE(results.borrow()) == 1) {
854              PyObject* result = PyTuple_GET_ITEM(results.borrow(), 0);
855              assert(result);
856              return OwnedObject::owning(result);
