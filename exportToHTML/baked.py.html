10   """Baked query extension. 
11    
12   Provides a creational pattern for the :class:`.query.Query` object which 
13   allows the fully constructed object, Core select statement, and string 
14   compiled result to be fully cached. 
15    
16    
17   """

34   class Bakery:
35       """Callable which returns a :class:`.BakedQuery`. 
36    
37       This object is returned by the class method 
38       :meth:`.BakedQuery.bakery`.  It exists as an object 
39       so that the "cache" can be easily inspected. 
40    
41       .. versionadded:: 1.2 
42    
43    
44       """
45   
46       __slots__ = "cls", "cache"
47   
48       def __init__(self, cls_, cache):
49           self.cls = cls_
50           self.cache = cache
51   
52       def __call__(self, initial_fn, *args):
53           return self.cls(self.cache, initial_fn, args)
54   
55   
56   class BakedQuery:
57       """A builder object for :class:`.query.Query` objects."""
58   
59       __slots__ = "steps", "_bakery", "_cache_key", "_spoiled"
60   
61       def __init__(self, bakery, initial_fn, args=()):
62           self._cache_key = ()
63           self._update_cache_key(initial_fn, args)
64           self.steps = [initial_fn]
65           self._spoiled = False
66           self._bakery = bakery
67   
68       @classmethod
69       def bakery(cls, size=200, _size_alert=None):
70           """Construct a new bakery. 
71    
72           :return: an instance of :class:`.Bakery` 
73    
74           """
75   
76           return Bakery(cls, util.LRUCache(size, size_alert=_size_alert))
77   
78       def _clone(self):
79           b1 = BakedQuery.__new__(BakedQuery)
80           b1._cache_key = self._cache_key
81           b1.steps = list(self.steps)
82           b1._bakery = self._bakery
83           b1._spoiled = self._spoiled
84           return b1
85   
86       def _update_cache_key(self, fn, args=()):
87           self._cache_key += (fn.__code__,) + args
88   
89       def __iadd__(self, other):
90           if isinstance(other, tuple):
91               self.add_criteria(*other)
92           else:
93               self.add_criteria(other)
94           return self
95   
96       def __add__(self, other):
97           if isinstance(other, tuple):
98               return self.with_criteria(*other)
99           else:
100              return self.with_criteria(other)
101  
102      def add_criteria(self, fn, *args):
103          """Add a criteria function to this :class:`.BakedQuery`. 
104   
105          This is equivalent to using the ``+=`` operator to 
106          modify a :class:`.BakedQuery` in-place. 
107   
108          """
109          self._update_cache_key(fn, args)
110          self.steps.append(fn)
111          return self
112  
113      def with_criteria(self, fn, *args):
114          """Add a criteria function to a :class:`.BakedQuery` cloned from this 
115          one. 
116   
117          This is equivalent to using the ``+`` operator to 
118          produce a new :class:`.BakedQuery` with modifications. 
119   
120          """
121          return self._clone().add_criteria(fn, *args)
122  
123      def for_session(self, session):
124          """Return a :class:`_baked.Result` object for this 
125          :class:`.BakedQuery`. 
126   
127          This is equivalent to calling the :class:`.BakedQuery` as a 
128          Python callable, e.g. ``result = my_baked_query(session)``. 
129   
130          """
131          return Result(self, session)
132  
133      def __call__(self, session):
134          return self.for_session(session)
135  
136      def spoil(self, full=False):
137          """Cancel any query caching that will occur on this BakedQuery object. 
138   
139          The BakedQuery can continue to be used normally, however additional 
140          creational functions will not be cached; they will be called 
141          on every invocation. 
142   
143          This is to support the case where a particular step in constructing 
144          a baked query disqualifies the query from being cacheable, such 
145          as a variant that relies upon some uncacheable value. 
146   
147          :param full: if False, only functions added to this 
148           :class:`.BakedQuery` object subsequent to the spoil step will be 
149           non-cached; the state of the :class:`.BakedQuery` up until 
150           this point will be pulled from the cache.   If True, then the 
151           entire :class:`_query.Query` object is built from scratch each 
152           time, with all creational functions being called on each 
153           invocation. 
154   
155          """
156          if not full and not self._spoiled:
157              _spoil_point = self._clone()
158              _spoil_point._cache_key += ("_query_only",)
159              self.steps = [_spoil_point._retrieve_baked_query]
160          self._spoiled = True
161          return self
162  
163      def _effective_key(self, session):
164          """Return the key that actually goes into the cache dictionary for 
165          this :class:`.BakedQuery`, taking into account the given 
166          :class:`.Session`. 
167   
168          This basically means we also will include the session's query_class, 
169          as the actual :class:`_query.Query` object is part of what's cached 
170          and needs to match the type of :class:`_query.Query` that a later 
171          session will want to use. 
172   
173          """
174          return self._cache_key + (session._query_cls,)
175  
176      def _with_lazyload_options(self, options, effective_path, cache_path=None):
177          """Cloning version of _add_lazyload_options."""
178          q = self._clone()
179          q._add_lazyload_options(options, effective_path, cache_path=cache_path)
180          return q
181  
182      def _add_lazyload_options(self, options, effective_path, cache_path=None):
183          """Used by per-state lazy loaders to add options to the 
184          "lazy load" query from a parent query. 
185   
186          Creates a cache key based on given load path and query options; 
187          if a repeatable cache key cannot be generated, the query is 
188          "spoiled" so that it won't use caching. 
189   
190          """
191  
192          key = ()
193  
194          if not cache_path:
195              cache_path = effective_path
196  
197          for opt in options:
198              if opt._is_legacy_option or opt._is_compile_state:
199                  ck = opt._generate_cache_key()
200                  if ck is None:
201                      self.spoil(full=True)
202                  else:
203                      assert not ck[1], (
204                          "loader options with variable bound parameters "
205                          "not supported with baked queries.  Please "
206                          "use new-style select() statements for cached "
207                          "ORM queries."
208                      )
209                      key += ck[0]
210  
211          self.add_criteria(
212              lambda q: q._with_current_path(effective_path).options(*options),
213              cache_path.path,
214              key,
215          )
216  
217      def _retrieve_baked_query(self, session):
218          query = self._bakery.get(self._effective_key(session), None)
219          if query is None:
220              query = self._as_query(session)
221              self._bakery[self._effective_key(session)] = query.with_session(
222                  None
223              )
224          return query.with_session(session)
225  
226      def _bake(self, session):
227          query = self._as_query(session)
228          query.session = None
229  
230          # in 1.4, this is where before_compile() event is
231          # invoked
232          statement = query._statement_20()
233  
234          # if the query is not safe to cache, we still do everything as though
235          # we did cache it, since the receiver of _bake() assumes subqueryload
236          # context was set up, etc.
237          #
238          # note also we want to cache the statement itself because this
239          # allows the statement itself to hold onto its cache key that is
240          # used by the Connection, which in itself is more expensive to
241          # generate than what BakedQuery was able to provide in 1.3 and prior
242  
243          if statement._compile_options._bake_ok:
244              self._bakery[self._effective_key(session)] = (
245                  query,
246                  statement,
247              )
248  
249          return query, statement
250  
251      def to_query(self, query_or_session):
252          """Return the :class:`_query.Query` object for use as a subquery. 
253   
254          This method should be used within the lambda callable being used 
255          to generate a step of an enclosing :class:`.BakedQuery`.   The 
256          parameter should normally be the :class:`_query.Query` object that 
257          is passed to the lambda:: 
258   
259              sub_bq = self.bakery(lambda s: s.query(User.name)) 
260              sub_bq += lambda q: q.filter(User.id == Address.user_id).correlate(Address) 
261   
262              main_bq = self.bakery(lambda s: s.query(Address)) 
263              main_bq += lambda q: q.filter(sub_bq.to_query(q).exists()) 
264   
265          In the case where the subquery is used in the first callable against 
266          a :class:`.Session`, the :class:`.Session` is also accepted:: 
267   
268              sub_bq = self.bakery(lambda s: s.query(User.name)) 
269              sub_bq += lambda q: q.filter(User.id == Address.user_id).correlate(Address) 
270   
271              main_bq = self.bakery( 
272                  lambda s: s.query(Address.id, sub_bq.to_query(q).scalar_subquery()) 
273              ) 
274   
275          :param query_or_session: a :class:`_query.Query` object or a class 
276           :class:`.Session` object, that is assumed to be within the context 
277           of an enclosing :class:`.BakedQuery` callable. 
278   
279   
280           .. versionadded:: 1.3 
281   
282   
283          """  # noqa: E501
284  
285          if isinstance(query_or_session, Session):
286              session = query_or_session
287          elif isinstance(query_or_session, Query):
288              session = query_or_session.session
289              if session is None:
290                  raise sa_exc.ArgumentError(
291                      "Given Query needs to be associated with a Session"
292                  )
293          else:
294              raise TypeError(
295                  "Query or Session object expected, got %r."
296                  % type(query_or_session)
297              )
298          return self._as_query(session)
299  
300      def _as_query(self, session):
301          query = self.steps[0](session)
302  
303          for step in self.steps[1:]:
304              query = step(query)
305  
306          return query

309  class Result:
310      """Invokes a :class:`.BakedQuery` against a :class:`.Session`. 
311   
312      The :class:`_baked.Result` object is where the actual :class:`.query.Query` 
313      object gets created, or retrieved from the cache, 
314      against a target :class:`.Session`, and is then invoked for results. 
315   
316      """
317  
318      __slots__ = "bq", "session", "_params", "_post_criteria"
319  
320      def __init__(self, bq, session):
321          self.bq = bq
322          self.session = session
323          self._params = {}
324          self._post_criteria = []
325  
326      def params(self, *args, **kw):
327          """Specify parameters to be replaced into the string SQL statement."""
328  
329          if len(args) == 1:
330              kw.update(args[0])
331          elif len(args) > 0:
332              raise sa_exc.ArgumentError(
333                  "params() takes zero or one positional argument, "
334                  "which is a dictionary."
335              )
336          self._params.update(kw)
337          return self
338  
339      def _using_post_criteria(self, fns):
340          if fns:
341              self._post_criteria.extend(fns)
342          return self
343  
344      def with_post_criteria(self, fn):
345          """Add a criteria function that will be applied post-cache. 
346   
347          This adds a function that will be run against the 
348          :class:`_query.Query` object after it is retrieved from the 
349          cache.    This currently includes **only** the 
350          :meth:`_query.Query.params` and :meth:`_query.Query.execution_options` 
351          methods. 
352   
353          .. warning::  :meth:`_baked.Result.with_post_criteria` 
354             functions are applied 
355             to the :class:`_query.Query` 
356             object **after** the query's SQL statement 
357             object has been retrieved from the cache.   Only 
358             :meth:`_query.Query.params` and 
359             :meth:`_query.Query.execution_options` 
360             methods should be used. 
361   
362   
363          .. versionadded:: 1.2 
364   
365   
366          """
367          return self._using_post_criteria([fn])
368  
369      def _as_query(self):
370          q = self.bq._as_query(self.session).params(self._params)
371          for fn in self._post_criteria:
372              q = fn(q)
373          return q
374  
375      def __str__(self):
376          return str(self._as_query())
377  
378      def __iter__(self):
379          return self._iter().__iter__()
380  
381      def _iter(self):
382          bq = self.bq
383  
384          if not self.session.enable_baked_queries or bq._spoiled:
385              return self._as_query()._iter()
386  
387          query, statement = bq._bakery.get(
388              bq._effective_key(self.session), (None, None)
389          )
390          if query is None:
391              query, statement = bq._bake(self.session)
392  
393          if self._params:
394              q = query.params(self._params)
395          else:
396              q = query
397          for fn in self._post_criteria:
398              q = fn(q)
399  
400          params = q._params
401          execution_options = dict(q._execution_options)
402          execution_options.update(
403              {
404                  "_sa_orm_load_options": q.load_options,
405                  "compiled_cache": bq._bakery,
406              }
407          )
408  
409          result = self.session.execute(
410              statement, params, execution_options=execution_options
411          )
412          if result._attributes.get("is_single_entity", False):
413              result = result.scalars()
414  
415          if result._attributes.get("filtered", False):
416              result = result.unique()
417  
418          return result
419  
420      def count(self):
421          """return the 'count'. 
422   
423          Equivalent to :meth:`_query.Query.count`. 
424   
425          Note this uses a subquery to ensure an accurate count regardless 
426          of the structure of the original statement. 
427   
428          """
429  
430          col = func.count(literal_column("*"))
431          bq = self.bq.with_criteria(lambda q: q._legacy_from_self(col))
432          return bq.for_session(self.session).params(self._params).scalar()
433  
434      def scalar(self):
435          """Return the first element of the first result or None 
436          if no rows present.  If multiple rows are returned, 
437          raises MultipleResultsFound. 
438   
439          Equivalent to :meth:`_query.Query.scalar`. 
440   
441          """
442          try:
443              ret = self.one()
444              if not isinstance(ret, collections_abc.Sequence):
445                  return ret
446              return ret[0]
447          except orm_exc.NoResultFound:
448              return None
449  
450      def first(self):
451          """Return the first row. 
452   
453          Equivalent to :meth:`_query.Query.first`. 
454   
455          """
456  
457          bq = self.bq.with_criteria(lambda q: q.slice(0, 1))
458          return (
459              bq.for_session(self.session)
460              .params(self._params)
461              ._using_post_criteria(self._post_criteria)
462              ._iter()
463              .first()
464          )
465  
466      def one(self):
467          """Return exactly one result or raise an exception. 
468   
469          Equivalent to :meth:`_query.Query.one`. 
470   
471          """
472          return self._iter().one()
473  
474      def one_or_none(self):
475          """Return one or zero results, or raise an exception for multiple 
476          rows. 
477   
478          Equivalent to :meth:`_query.Query.one_or_none`. 
479   
480          """
481          return self._iter().one_or_none()
482  
483      def all(self):
484          """Return all rows. 
485   
486          Equivalent to :meth:`_query.Query.all`. 
487   
488          """
489          return self._iter().all()
490  
491      def get(self, ident):
492          """Retrieve an object based on identity. 
493   
494          Equivalent to :meth:`_query.Query.get`. 
495   
496          """
497  
498          query = self.bq.steps[0](self.session)
499          return query._get_impl(ident, self._load_on_pk_identity)
500  
501      def _load_on_pk_identity(self, session, query, primary_key_identity, **kw):
502          """Load the given primary key identity from the database."""
503  
504          mapper = query._raw_columns[0]._annotations["parententity"]
505  
506          _get_clause, _get_params = mapper._get_clause
507  
508          def setup(query):
509              _lcl_get_clause = _get_clause
510              q = query._clone()
511              q._get_condition()
512              q._order_by = None
513  
514              # None present in ident - turn those comparisons
515              # into "IS NULL"
516              if None in primary_key_identity:
517                  nones = {
518                      _get_params[col].key
519                      for col, value in zip(
520                          mapper.primary_key, primary_key_identity
521                      )
522                      if value is None
523                  }
524                  _lcl_get_clause = sql_util.adapt_criterion_to_null(
525                      _lcl_get_clause, nones
526                  )
527  
528              # TODO: can mapper._get_clause be pre-adapted?
529              q._where_criteria = (
530                  sql_util._deep_annotate(_lcl_get_clause, {"_orm_adapt": True}),
531              )
532  
533              for fn in self._post_criteria:
534                  q = fn(q)
535              return q
536  
537          # cache the query against a key that includes
538          # which positions in the primary key are NULL
539          # (remember, we can map to an OUTER JOIN)
540          bq = self.bq
541  
542          # add the clause we got from mapper._get_clause to the cache
543          # key so that if a race causes multiple calls to _get_clause,
544          # we've cached on ours
545          bq = bq._clone()
546          bq._cache_key += (_get_clause,)
547  
548          bq = bq.with_criteria(
549              setup, tuple(elem is None for elem in primary_key_identity)
550          )
551  
552          params = {
553              _get_params[primary_key].key: id_val
554              for id_val, primary_key in zip(
555                  primary_key_identity, mapper.primary_key
556              )
557          }
558  
559          result = list(bq.for_session(self.session).params(**params))
560          l = len(result)
561          if l > 1:
562              raise orm_exc.MultipleResultsFound()
563          elif l:
564              return result[0]
565          else:
566              return None
