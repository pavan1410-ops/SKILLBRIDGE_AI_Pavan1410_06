51   static int
52   slp_switch(void)
53   {
54       int err;
55       int *stackref, stsizediff;
56   
57       /* Put current stack pointer into stackref. 
58        * Register spilling is done in save/restore. 
59        */
60       __asm__ volatile ("mov %%sp, %0" : "=r" (stackref));
61   
62       {
63           /* Thou shalt put SLP_SAVE_STATE into a local block */
64           /* Copy the current stack onto the heap */
65           SLP_SAVE_STATE(stackref, stsizediff);
66   
67           /* Increment stack and frame pointer by stsizediff */
68           __asm__ volatile (
69               "add %0, %%sp, %%sp\n\t"
70               "add %0, %%fp, %%fp"
71               : : "r" (stsizediff));
72   
73           /* Copy new stack from it's save store on the heap */
74           SLP_RESTORE_STATE();
75   
76           __asm__ volatile ("mov %1, %0" : "=r" (err) : "i" (0));
77           return err;
78       }
79   }
