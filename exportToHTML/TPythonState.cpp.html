9    PythonState::PythonState()
10       : _top_frame()
11   #if GREENLET_USE_CFRAME
12       ,cframe(nullptr)
13       ,use_tracing(0)
14   #endif
15   #if GREENLET_PY314
16       ,py_recursion_depth(0)
17       ,current_executor(nullptr)
18       ,stackpointer(nullptr)
19       #ifdef Py_GIL_DISABLED
20       ,c_stack_refs(nullptr)
21       #endif
22   #elif GREENLET_PY312
23       ,py_recursion_depth(0)
24       ,c_recursion_depth(0)
25   #else
26       ,recursion_depth(0)
27   #endif
28   #if GREENLET_PY313
29       ,delete_later(nullptr)
30       ,critical_section(0)
31   #else
32       ,trash_delete_nesting(0)
33   #endif
34   #if GREENLET_PY311
35       ,current_frame(nullptr)
36       ,datastack_chunk(nullptr)
37       ,datastack_top(nullptr)
38       ,datastack_limit(nullptr)
39   #endif
40   {
41   #if GREENLET_USE_CFRAME
42       /* 
43         The PyThreadState->cframe pointer usually points to memory on 
44         the stack, alloceted in a call into PyEval_EvalFrameDefault. 
45    
46         Initially, before any evaluation begins, it points to the 
47         initial PyThreadState object's ``root_cframe`` object, which is 
48         statically allocated for the lifetime of the thread. 
49    
50         A greenlet can last for longer than a call to 
51         PyEval_EvalFrameDefault, so we can't set its ``cframe`` pointer 
52         to be the current ``PyThreadState->cframe``; nor could we use 
53         one from the greenlet parent for the same reason. Yet a further 
54         no: we can't allocate one scoped to the greenlet and then 
55         destroy it when the greenlet is deallocated, because inside the 
56         interpreter the _PyCFrame objects form a linked list, and that too 
57         can result in accessing memory beyond its dynamic lifetime (if 
58         the greenlet doesn't actually finish before it dies, its entry 
59         could still be in the list). 
60    
61         Using the ``root_cframe`` is problematic, though, because its 
62         members are never modified by the interpreter and are set to 0, 
63         meaning that its ``use_tracing`` flag is never updated. We don't 
64         want to modify that value in the ``root_cframe`` ourself: it 
65         *shouldn't* matter much because we should probably never get 
66         back to the point where that's the only cframe on the stack; 
67         even if it did matter, the major consequence of an incorrect 
68         value for ``use_tracing`` is that if its true the interpreter 
69         does some extra work --- however, it's just good code hygiene. 
70    
71         Our solution: before a greenlet runs, after its initial 
72         creation, it uses the ``root_cframe`` just to have something to 
73         put there. However, once the greenlet is actually switched to 
74         for the first time, ``g_initialstub`` (which doesn't actually 
75         "return" while the greenlet is running) stores a new _PyCFrame on 
76         its local stack, and copies the appropriate values from the 
77         currently running _PyCFrame; this is then made the _PyCFrame for the 
78         newly-minted greenlet. ``g_initialstub`` then proceeds to call 
79         ``glet.run()``, which results in ``PyEval_...`` adding the 
80         _PyCFrame to the list. Switches continue as normal. Finally, when 
81         the greenlet finishes, the call to ``glet.run()`` returns and 
82         the _PyCFrame is taken out of the linked list and the stack value 
83         is now unused and free to expire. 
84    
85         XXX: I think we can do better. If we're deallocing in the same 
86         thread, can't we traverse the list and unlink our frame? 
87         Can we just keep a reference to the thread state in case we 
88         dealloc in another thread? (Is that even possible if we're still 
89         running and haven't returned from g_initialstub?) 
90       */
91       this->cframe = &PyThreadState_GET()->root_cframe;
92   #endif
93   }

95   #if GREENLET_PY314 && defined(Py_GIL_DISABLED)
96   void PythonState::capture_c_stack_refs(const PyThreadState* tstate) noexcept
97   {
98       // Runs from operator<< while our C stack is still live and coherent, so we
99       // can walk tstate's _PyCStackRef list and take a strong reference to every
100      // object it holds. tp_traverse visits these once we're suspended, because
101      // by then the nodes themselves have been relocated into the heap stack copy
102      // and the saved list head no longer points at them. Strong references (not
103      // _Py_VISIT_STACKREF, whose _PyGC_VisitStackRef isn't exported before 3.15);
104      // a std::vector rather than a Python list/tuple because operator<< must not
105      // allocate a GC-tracked object mid-switch. Rebuilt from scratch each time;
106      // the list is empty at a typical switch, so this is usually just an empty
107      // loop.
108      this->c_stack_ref_snapshot.clear();
109      for (const _PyCStackRef* node = ((_PyThreadStateImpl*)tstate)->c_stack_refs;
110           node != nullptr; node = node->next) {
111          if (!PyStackRef_IsNullOrInt(node->ref)) {
112              this->c_stack_ref_snapshot.push_back(
113                  OwnedObject::owning(PyStackRef_AsPyObjectBorrow(node->ref)));
114          }
115      }
116  }
117  #endif

120  inline void PythonState::may_switch_away() noexcept
121  {
122  #if GREENLET_PY311
123      // PyThreadState_GetFrame is probably going to have to allocate a
124      // new frame object. That may trigger garbage collection. Because
125      // we call this during the early phases of a switch (it doesn't
126      // matter to which greenlet, as this has a global effect), if a GC
127      // triggers a switch away, two things can happen, both bad:
128      // - We might not get switched back to, halting forward progress.
129      //   this is pathological, but possible.
130      // - We might get switched back to with a different set of
131      //   arguments or a throw instead of a switch. That would corrupt
132      //   our state (specifically, PyErr_Occurred() and this->args()
133      //   would no longer agree).
134      //
135      // Thus, when we call this API, we need to have GC disabled.
136      // This method serves as a bottleneck we call when maybe beginning
137      // a switch. In this way, it is always safe -- no risk of GC -- to
138      // use ``_GetFrame()`` whenever we need to, just as it was in
139      // <=3.10 (because subsequent calls will be cached and not
140      // allocate memory).
141  
142      GCDisabledGuard no_gc;
143      Py_XDECREF(PyThreadState_GetFrame(PyThreadState_GET()));
144  #endif
145  }

147  void PythonState::operator<<(const PyThreadState *const tstate) noexcept
148  {
149      this->_context.steal(tstate->context);
150  #if GREENLET_USE_CFRAME
151      /* 
152        IMPORTANT: ``cframe`` is a pointer into the STACK. Thus, because 
153        the call to ``slp_switch()`` changes the contents of the stack, 
154        you cannot read from ``ts_current->cframe`` after that call and 
155        necessarily get the same values you get from reading it here. 
156        Anything you need to restore from now to then must be saved in a 
157        global/threadlocal variable (because we can't use stack 
158        variables here either). For things that need to persist across 
159        the switch, use `will_switch_from`. 
160      */
161      this->cframe = tstate->cframe;
162    #if !GREENLET_PY312
163      this->use_tracing = tstate->cframe->use_tracing;
164    #endif
165  #endif // GREENLET_USE_CFRAME
166  #if GREENLET_PY311
167    #if GREENLET_PY314
168      this->py_recursion_depth = tstate->py_recursion_limit - tstate->py_recursion_remaining;
169      this->current_executor = tstate->current_executor;
170      #ifdef Py_GIL_DISABLED
171      this->c_stack_refs = ((_PyThreadStateImpl*)tstate)->c_stack_refs;
172      // Capture the deferred references now, while our C stack is still live, so
173      // tp_traverse can keep them from being collected while we're suspended.
174      this->capture_c_stack_refs(tstate);
175      #endif
176    #elif GREENLET_PY312
177      this->py_recursion_depth = tstate->py_recursion_limit - tstate->py_recursion_remaining;
178      this->c_recursion_depth = Py_C_RECURSION_LIMIT - tstate->c_recursion_remaining;
179    #else // not 312
180      this->recursion_depth = tstate->recursion_limit - tstate->recursion_remaining;
181    #endif // GREENLET_PY312
182    #if GREENLET_PY313
183      this->current_frame = tstate->current_frame;
184    #elif GREENLET_USE_CFRAME
185      this->current_frame = tstate->cframe->current_frame;
186    #endif
187      this->datastack_chunk = tstate->datastack_chunk;
188      this->datastack_top = tstate->datastack_top;
189      this->datastack_limit = tstate->datastack_limit;
190  
191      PyFrameObject *frame = PyThreadState_GetFrame((PyThreadState *)tstate);
192      Py_XDECREF(frame);  // PyThreadState_GetFrame gives us a new
193                          // reference.
194      this->_top_frame.steal(frame);
195    #if GREENLET_PY314
196      if (this->top_frame()) {
197          this->stackpointer = this->_top_frame->f_frame->stackpointer;
198      }
199      else {
200          this->stackpointer = nullptr;
201      }
202    #endif
203    #if GREENLET_PY313
204      // By contract of _PyTrash_thread_deposit_object,
205      // the ``delete_later`` object has a refcount of 0.
206      // We take a strong reference to it.
207      //
208      // Now, ``delete_later`` is managed as a
209      // linked list whose objects are unconditionally deallocated
210      // WITHOUT calling DECREF on them, so it's not clear what that is
211      // actually accomplishing. That is, if another object is pushed on
212      // the list and then the list is deallocated, this object will
213      // still be deallocated. This strong reference serves as a form of
214      // resurrection, meaning that when operator>> DECREFs it, we might
215      // enter its ``tp_dealloc`` function again.
216      //
217      // In practice, it's quite difficult to arrange for this to be
218      // a non-null value during a greenlet switch.
219      // ``greenlet.tests.test_greenlet_trash`` tries, but under 3.14,
220      // at least, fails to do so.
221      this->delete_later = Py_XNewRef(tstate->delete_later);
222  #ifdef Py_GIL_DISABLED
223      // Switching greenlets swaps C stacks, which to the free-threaded runtime is
224      // the same predicament as detaching the thread: the PyCriticalSection nodes
225      // chained off tstate->critical_section live on the stack we're leaving, and
226      // their PyMutexes would stay locked behind our back. The greenlet we switch
227      // to could then block forever taking one of those same locks -- e.g. an
228      // asyncio event dispatched onto another fiber re-enters a Task/Future that
229      // the suspended fiber is mid-step on. So drop the locks here the way
230      // _PyThreadState_Detach() does and let operator>> re-take them on resume.
231      if (tstate->critical_section != 0) {
232          _PyCriticalSection_SuspendAll(const_cast<PyThreadState*>(tstate));
233      }
234  #endif
235      this->critical_section = tstate->critical_section;
236    #elif GREENLET_PY312
237      this->trash_delete_nesting = tstate->trash.delete_nesting;
238    #else // not 312 or 3.13+
239      this->trash_delete_nesting = tstate->trash_delete_nesting;
240    #endif // GREENLET_PY312
241  #else // Not 311
242      this->recursion_depth = tstate->recursion_depth;
243      this->_top_frame.steal(tstate->frame);
244      this->trash_delete_nesting = tstate->trash_delete_nesting;
245  #endif // GREENLET_PY311
246  }

248  #if GREENLET_PY312
249  void GREENLET_NOINLINE(PythonState::unexpose_frames)()
250  {
251      if (!this->top_frame()) {
252          return;
253      }
254  
255      // See GreenletState::expose_frames() and the comment on frames_were_exposed
256      // for more information about this logic.
257      _PyInterpreterFrame *iframe = this->_top_frame->f_frame;
258      while (iframe != nullptr) {
259          _PyInterpreterFrame *prev_exposed = iframe->previous;
260          assert(iframe->frame_obj);
261          memcpy(&iframe->previous, &iframe->frame_obj->_f_frame_data[0],
262                 sizeof(void *));
263          iframe = prev_exposed;
264      }
265  }

267  void PythonState::unexpose_frames()
268  {}
269  #endif

271  void PythonState::operator>>(PyThreadState *const tstate) noexcept
272  {
273      tstate->context = this->_context.relinquish_ownership();
274      /* Incrementing this value invalidates the contextvars cache, 
275         which would otherwise remain valid across switches */
276      tstate->context_ver++;
277  #if GREENLET_USE_CFRAME
278      tstate->cframe = this->cframe;
279      /* 
280        If we were tracing, we need to keep tracing. 
281        There should never be the possibility of hitting the 
282        root_cframe here. See note above about why we can't 
283        just copy this from ``origin->cframe->use_tracing``. 
284      */
285    #if !GREENLET_PY312
286      tstate->cframe->use_tracing = this->use_tracing;
287    #endif
288  #endif // GREENLET_USE_CFRAME
289  #if GREENLET_PY311
290    #if GREENLET_PY314
291      tstate->py_recursion_remaining = tstate->py_recursion_limit - this->py_recursion_depth;
292      tstate->current_executor = this->current_executor;
293      #ifdef Py_GIL_DISABLED
294      ((_PyThreadStateImpl*)tstate)->c_stack_refs = this->c_stack_refs;
295      // We're the running greenlet again: our C-stack refs live in the thread
296      // state now and gc_visit_thread_stacks() covers them, so drop the strong
297      // references tp_traverse held on our behalf while we were suspended.
298      this->c_stack_ref_snapshot.clear();
299      #endif
300      this->unexpose_frames();
301    #elif GREENLET_PY312
302      tstate->py_recursion_remaining = tstate->py_recursion_limit - this->py_recursion_depth;
303      tstate->c_recursion_remaining = Py_C_RECURSION_LIMIT - this->c_recursion_depth;
304      this->unexpose_frames();
305    #else // \/ 3.11
306      tstate->recursion_remaining = tstate->recursion_limit - this->recursion_depth;
307    #endif // GREENLET_PY312
308    #if GREENLET_PY313
309      tstate->current_frame = this->current_frame;
310    #elif GREENLET_USE_CFRAME
311      tstate->cframe->current_frame = this->current_frame;
312    #endif
313      tstate->datastack_chunk = this->datastack_chunk;
314      tstate->datastack_top = this->datastack_top;
315      tstate->datastack_limit = this->datastack_limit;
316  #if GREENLET_PY314 && defined(Py_GIL_DISABLED)
317      if (this->top_frame()) {
318          this->_top_frame->f_frame->stackpointer = this->stackpointer;
319      }
320  #endif
321      this->_top_frame.relinquish_ownership();
322    #if GREENLET_PY313
323      // See comments in operator<<. We own a strong reference to
324      // this->delete_later, which may or may not be the same object as
325      // tstate->delete_later (depending if something pushed an object
326      // onto the trashcan). Again, because ``delete_later`` is managed
327      // as a linked list, it's not clear that saving and restoring the
328      // value, especially without ever setting it to NULL, accomplishes
329      // much...but the code was added by a core dev, so assume correct.
330      //
331      // Recall that tstate->delete_later is supposed to have a refcount
332      // of 0, because objects are added there from their ``tp_dealloc``
333      // method. So we should only need to DECREF it if we're the ones
334      // that INCREF'd it in operator<<. (This is different than the
335      // core dev's original code which always did this.)
336      if (this->delete_later == tstate->delete_later) {
337          Py_XDECREF(tstate->delete_later);
338          tstate->delete_later = this->delete_later;
339          this->delete_later = nullptr;
340      }
341      else {
342          // it got switched behind our back. So the reference we own
343          // needs to be explicitly cleared.
344          tstate->delete_later = this->delete_later;
345          Py_CLEAR(this->delete_later);
346      }
347      tstate->critical_section = this->critical_section;
348  #ifdef Py_GIL_DISABLED
349      // Re-acquire whatever operator<< suspended when this greenlet last yielded.
350      // A no-op for a greenlet that held no locks, and for a brand-new one whose
351      // chain starts empty. Mirrors the resume in _PyThreadState_Attach(); note
352      // _PyCriticalSection_Resume() dereferences the head, so the != 0 guard is
353      // load-bearing, not just a fast path.
354      if (tstate->critical_section != 0) {
355          _PyCriticalSection_Resume(tstate);
356      }
357  #endif
358  
359    #elif GREENLET_PY312
360      tstate->trash.delete_nesting = this->trash_delete_nesting;
361    #else // not 3.12
362      tstate->trash_delete_nesting = this->trash_delete_nesting;
363    #endif // GREENLET_PY312
364  #else // not 3.11
365      tstate->frame = this->_top_frame.relinquish_ownership();
366      tstate->recursion_depth = this->recursion_depth;
367      tstate->trash_delete_nesting = this->trash_delete_nesting;
368  #endif // GREENLET_PY311
369  }

371  inline void PythonState::will_switch_from(PyThreadState *const origin_tstate) noexcept
372  {
373  #if GREENLET_USE_CFRAME && !GREENLET_PY312
374      // The weird thing is, we don't actually save this for an
375      // effect on the current greenlet, it's saved for an
376      // effect on the target greenlet. That is, we want
377      // continuity of this setting across the greenlet switch.
378      this->use_tracing = origin_tstate->cframe->use_tracing;
379  #endif

382  void PythonState::set_initial_state(const PyThreadState* const tstate) noexcept
383  {
384      this->_top_frame = nullptr;
385  #if GREENLET_PY314
386      this->py_recursion_depth = tstate->py_recursion_limit - tstate->py_recursion_remaining;
387      this->current_executor = tstate->current_executor;
388      #ifdef Py_GIL_DISABLED
389      // Start with an empty C-stack-ref list, the way a brand-new thread does;
390      // do NOT copy the parent thread state's head. Those _PyCStackRef nodes sit
391      // on the parent greenlet's C stack, so once we start running on our own
392      // stack and overwrite that region, following them reads garbage. The
393      // free-threaded collector walks c_stack_refs for every thread in
394      // gc_visit_thread_stacks(), so leaving the stale head here crashed it.
395      // See https://github.com/python-greenlet/greenlet/issues/515.
396      this->c_stack_refs = nullptr;
397      #endif
398      // this->stackpointer is left null because this->_top_frame is
399      // null so there is no value to copy.
400  #elif GREENLET_PY312
401      this->py_recursion_depth = tstate->py_recursion_limit - tstate->py_recursion_remaining;
402  #if GREENLET_314
403      this->c_recursion_depth = 0; // unused on 3.14
404  #else
405      this->c_recursion_depth = Py_C_RECURSION_LIMIT - tstate->c_recursion_remaining;
406  #endif
407  #elif GREENLET_PY311
408      this->recursion_depth = tstate->recursion_limit - tstate->recursion_remaining;
409  #else
410      this->recursion_depth = tstate->recursion_depth;
411  #endif
412  }
413  // TODO: Better state management about when we own the top frame.
414  int PythonState::tp_traverse(visitproc visit, void* arg, bool visit_top_frame) noexcept
415  {
416      Py_VISIT(this->_context.borrow());
417      if (visit_top_frame) {
418          Py_VISIT(this->_top_frame.borrow());
419      }
420  #if GREENLET_PY315
421      // Visit the references held by our suspended frames.
422      // This is important specially on free-threading where the
423      // the suspended frames may contain deferred references to
424      // objects, and if they are not traversed then the interpreter
425      // can free objects early causing a use-after-free crash
426      // at runtime exit.
427      if (this->_top_frame) {
428          for (_PyInterpreterFrame* iframe = this->_top_frame->f_frame;
429               iframe != nullptr; iframe = iframe->previous) {
430              // Skip generator/coroutine frames; their object's traverse
431              // already visits them (gen_traverse), so we'd double-count.
432              // expose_frames leaves them in the ->previous chain.
433              if (iframe->owner != FRAME_OWNED_BY_THREAD) {
434                  continue;
435              }
436              Py_VISIT(iframe->frame_obj);
437              Py_VISIT(iframe->f_locals);
438              _Py_VISIT_STACKREF(iframe->f_funcobj);
439              _Py_VISIT_STACKREF(iframe->f_executable);
440              int frame_result = _PyGC_VisitFrameStack(iframe, visit, arg);
441              if (frame_result) {
442                  return frame_result;
443              }
444          }
445      }
446  #endif
447  #if GREENLET_PY314 && defined(Py_GIL_DISABLED)
448      // Visit the objects this greenlet's C-stack refs were holding when it
449      // suspended (captured by capture_c_stack_refs). The free-threaded collector
450      // only walks the running thread's _PyCStackRef list in
451      // gc_visit_thread_stacks(), so without this a collection could free an
452      // object reachable only through a suspended greenlet's C-stack ref and we'd
453      // use it after free once the greenlet resumed. The snapshot is empty while
454      // we're the running greenlet, so this is a no-op there.
455      for (const OwnedObject& ref : this->c_stack_ref_snapshot) {
456          Py_VISIT(ref.borrow());
457      }
458  #endif
459      // Note that we DO NOT visit ``delete_later``. Even if it's
460      // non-null and we technically own a reference to it, its
461      // reference count already went to 0 once and it was in the
462      // process of being deallocated. The trash can mechanism linked it
463      // into a list that will be cleaned at some later time, and it has
464      // become untracked by the GC.
465      return 0;
466  }

468  void PythonState::tp_clear(bool own_top_frame) noexcept
469  {
470      PythonStateContext::tp_clear();
471  #if GREENLET_PY314 && defined(Py_GIL_DISABLED)
472      this->c_stack_ref_snapshot.clear();
473  #endif
474      // If we get here owning a frame,
475      // we got dealloc'd without being finished. We may or may not be
476      // in the same thread.
477      if (own_top_frame) {
478  #if GREENLET_PY315
479          // Release the references held by our suspended frames.
480          // this->top_frame gets implicitly cleared by the Py_CLEAR(iframe->frame_obj)
481          // of the first complete frame, so in the end we relinquish ownership of it.
482          if (this->_top_frame) {
483              for (_PyInterpreterFrame* iframe = this->_top_frame->f_frame;
484                   iframe != nullptr; iframe = iframe->previous) {
485                  if (iframe->owner != FRAME_OWNED_BY_THREAD) {
486                      continue;
487                  }
488                  // Clear the references held by this frame's evaluation stack.
489                  _PyStackRef* locals = iframe->localsplus;
490                  _PyStackRef* sp = iframe->stackpointer;
491                  if (sp) {
492                      while (sp > locals) {
493                          sp--;
494                          PyStackRef_CLEAR(*sp);
495                      }
496                      iframe->stackpointer = locals;
497                  }
498                  Py_CLEAR(iframe->f_locals);
499                  Py_CLEAR(iframe->frame_obj);
500                  PyStackRef_CLEAR(iframe->f_funcobj);
501                  PyStackRef_CLEAR(iframe->f_executable);
502              }
503          }
504          this->_top_frame.relinquish_ownership();
505  #else
506          this->_top_frame.CLEAR();
507  #endif
508      }
509  }

530  void PythonState::did_finish(PyThreadState* tstate) noexcept
531  {
532  #if GREENLET_PY311
533      // See https://github.com/gevent/gevent/issues/1924 and
534      // https://github.com/python-greenlet/greenlet/issues/328. In
535      // short, Python 3.11 allocates memory for frames as a sort of
536      // linked list that's kept as part of PyThreadState in the
537      // ``datastack_chunk`` member and friends. These are saved and
538      // restored as part of switching greenlets.
539      //
540      // When we initially switch to a greenlet, we set those to NULL.
541      // That causes the frame management code to treat this like a
542      // brand new thread and start a fresh list of chunks, beginning
543      // with a new "root" chunk. As we make calls in this greenlet,
544      // those chunks get added, and as calls return, they get popped.
545      // But the frame code (pystate.c) is careful to make sure that the
546      // root chunk never gets popped.
547      //
548      // Thus, when a greenlet exits for the last time, there will be at
549      // least a single root chunk that we must be responsible for
550      // deallocating.
551      //
552      // The complex part is that these chunks are allocated and freed
553      // using ``_PyObject_VirtualAlloc``/``Free``. Those aren't public
554      // functions, and they aren't exported for linking. It so happens
555      // that we know they are just thin wrappers around the Arena
556      // allocator, so we can use that directly to deallocate in a
557      // compatible way.
558      //
559      // CAUTION: Check this implementation detail on every major version.
560      //
561      // It might be nice to be able to do this in our destructor, but
562      // can we be sure that no one else is using that memory? Plus, as
563      // described below, our pointers may not even be valid anymore. As
564      // a special case, there is one time that we know we can do this,
565      // and that's from the destructor of the associated UserGreenlet
566      // (NOT main greenlet)
567      PyObjectArenaAllocator alloc;
568      _PyStackChunk* chunk = nullptr;
569      if (tstate) {
570          // We really did finish, we can never be switched to again.
571          chunk = tstate->datastack_chunk;
572          // Unfortunately, we can't do much sanity checking. Our
573          // this->datastack_chunk pointer is out of date (evaluation may
574          // have popped down through it already) so we can't verify that
575          // we deallocate it. I don't think we can even check datastack_top
576          // for the same reason.
577  
578          PyObject_GetArenaAllocator(&alloc);
579          tstate->datastack_chunk = nullptr;
580          tstate->datastack_limit = nullptr;
581          tstate->datastack_top = nullptr;
582  
583      }
584      else if (this->datastack_chunk) {
585          // The UserGreenlet (NOT the main greenlet!) is being deallocated. If we're
586          // still holding a stack chunk, it's garbage because we know
587          // we can never switch back to let cPython clean it up.
588          // Because the last time we got switched away from, and we
589          // haven't run since then, we know our chain is valid and can
590          // be dealloced.
591          chunk = this->datastack_chunk;
592          PyObject_GetArenaAllocator(&alloc);
593      }
594  
595      if (alloc.free && chunk) {
596          // In case the arena mechanism has been torn down already.
597          while (chunk) {
598              _PyStackChunk *prev = chunk->previous;
599              chunk->previous = nullptr;
600              alloc.free(alloc.ctx, chunk, chunk->size);
601              chunk = prev;
602          }
603      }
604  
605      this->datastack_chunk = nullptr;
606      this->datastack_limit = nullptr;
607      this->datastack_top = nullptr;
608  #endif
609  }
