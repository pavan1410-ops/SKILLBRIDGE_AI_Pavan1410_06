29   static PyGreenlet*
30   PyGreenlet_GetCurrent(void)
31   {
32       if (greenlet::IsShuttingDown()) {
33           PyErr_SetString(PyExc_RuntimeError, "greenlet is being finalized");
34           return nullptr;
35       }
36       return GET_THREAD_STATE().state().get_current().relinquish_ownership();
37   }

39   static int
40   PyGreenlet_SetParent(PyGreenlet* g, PyGreenlet* nparent)
41   {
42       return green_setparent((PyGreenlet*)g, (PyObject*)nparent, NULL);
43   }

45   static PyGreenlet*
46   PyGreenlet_New(PyObject* run, PyGreenlet* parent)
47   {
48       using greenlet::refs::NewDictReference;
49       // In the past, we didn't use green_new and green_init, but that
50       // was a maintenance issue because we duplicated code. This way is
51       // much safer, but slightly slower. If that's a problem, we could
52       // refactor green_init to separate argument parsing from initialization.
53       OwnedGreenlet g = OwnedGreenlet::consuming(green_new(&PyGreenlet_Type, nullptr, nullptr));
54       if (!g) {
55           return NULL;
56       }
57   
58       try {
59           NewDictReference kwargs;
60           if (run) {
61               kwargs.SetItem(mod_globs->str_run, run);
62           }
63           if (parent) {
64               kwargs.SetItem("parent", (PyObject*)parent);
65           }
66   
67           Require(green_init(g.borrow(), mod_globs->empty_tuple, kwargs.borrow()));
68       }
69       catch (const PyErrOccurred&) {
70           return nullptr;
71       }
72   
73       return g.relinquish_ownership();
74   }
75   
76   static PyObject*
77   PyGreenlet_Switch(PyGreenlet* self, PyObject* args, PyObject* kwargs)
78   {
79       if (!PyGreenlet_Check(self)) {
80           PyErr_BadArgument();
81           return NULL;
82       }
83   
84       if (args == NULL) {
85           args = mod_globs->empty_tuple;
86       }
87   
88       if (kwargs == NULL || !PyDict_Check(kwargs)) {
89           kwargs = NULL;
90       }
91   
92       return green_switch(self, args, kwargs);
93   }
94   
95   static PyObject*
96   PyGreenlet_Throw(PyGreenlet* self, PyObject* typ, PyObject* val, PyObject* tb)
97   {
98       if (!PyGreenlet_Check(self)) {
99           PyErr_BadArgument();
100          return nullptr;
101      }
102      try {
103          PyErrPieces err_pieces(typ, val, tb);
104          return internal_green_throw(self, err_pieces).relinquish_ownership();
105      }
106      catch (const PyErrOccurred&) {
107          return nullptr;
108      }
109  }

113  static int
114  Extern_PyGreenlet_MAIN(PyGreenlet* self)
115  {
116      if (!PyGreenlet_Check(self)) {
117          PyErr_BadArgument();
118          return -1;
119      }
120      return self->pimpl->main();
121  }

143  static PyGreenlet*
144  Extern_PyGreenlet_GET_PARENT(PyGreenlet* self)
145  {
146      if (!PyGreenlet_Check(self)) {
