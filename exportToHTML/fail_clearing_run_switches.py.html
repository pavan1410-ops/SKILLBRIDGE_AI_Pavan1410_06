2    """ 
3    If we have a run callable passed to the constructor or set as an 
4    attribute, but we don't actually use that (because ``__getattribute__`` 
5    or the like interferes), then when we clear callable before beginning 
6    to run, there's an opportunity for Python code to run. 
7     
8    """
9    import greenlet
10   
11   g = None
12   main = greenlet.getcurrent()
13   
14   results = []
15   
16   class RunCallable:
17   
18       def __del__(self):
19           results.append(('RunCallable', '__del__'))
20           main.switch('from RunCallable')
21   
22   
23   class G(greenlet.greenlet):
24   
25       def __getattribute__(self, name):
26           if name == 'run':
27               results.append(('G.__getattribute__', 'run'))
28               return run_func
29           return object.__getattribute__(self, name)
30   
31   
32   def run_func():
33       results.append(('run_func', 'enter'))
34   
35   
36   g = G(RunCallable())
37   # Try to start G. It will get to the point where it deletes
38   # its run callable C++ variable in inner_bootstrap. That triggers
39   # the __del__ method, which switches back to main before g
40   # actually even starts running.
41   x = g.switch()
42   results.append(('main: g.switch()', x))
43   # In the C++ code, this results in g->g_switch() appearing to return, even though
44   # it has yet to run.
45   print('In main with', x, flush=True)
46   g.switch()
47   print('RESULTS', results)
