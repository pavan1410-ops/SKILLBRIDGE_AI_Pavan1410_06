17   #define REGS_TO_SAVE "$16", "$17", "$18", "$19", "$20", "$21", "$22", \
18          "$23", "$30"
19   __attribute__((nomips16))
20   static int
21   slp_switch(void)
22   {
23       int err;
24       int *stackref, stsizediff;
25   #ifdef __mips64
26       uint64_t gpsave;
27   #endif
28       __asm__ __volatile__ ("" : : : REGS_TO_SAVE);
29   #ifdef __mips64
30       __asm__ __volatile__ ("sd $28,%0" : "=m" (gpsave) : : );
31   #endif
32       __asm__ ("move %0, $29" : "=r" (stackref) : );
33       {
34           SLP_SAVE_STATE(stackref, stsizediff);
35           __asm__ __volatile__ (
36   #ifdef __mips64
37               "daddu $29, $29, %0\n"
38   #else
39               "addu $29, $29, %0\n"
40   #endif
41               : /* no outputs */
42               : "r" (stsizediff)
43               );
44           SLP_RESTORE_STATE();
45       }
46   #ifdef __mips64
47       __asm__ __volatile__ ("ld $28,%0" : : "m" (gpsave) : );
48   #endif
49       __asm__ __volatile__ ("" : : : REGS_TO_SAVE);
50       __asm__ __volatile__ ("move %0, $0" : "=r" (err));
51       return err;
52   }
