56   cdef class DecimalResultProcessor:
57       cdef object type_
58       cdef str format_
59   
60       def __cinit__(self, type_, format_):
61           self.type_ = type_
62           self.format_ = format_
63   
64       def process(self, object value):
65           if value is None:
66               return None
67           else:
68               return self.type_(self.format_ % value)
