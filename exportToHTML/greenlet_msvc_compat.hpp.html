3    /* 
4     * Support for MSVC on Windows. 
5     * 
6     * Beginning with Python 3.14, some of the internal 
7     * include files we need are not compatible with MSVC 
8     * in C++ mode: 
9     * 
10    *   internal\pycore_stackref.h(253): error C4576: a parenthesized type 
11    *    followed by an initializer list is a non-standard explicit type conversion syntax 
12    * 
13    * This file is included from ``internal/pycore_interpframe.h``, which 
14    * we need for the ``_PyFrame_IsIncomplete`` API. 
15    * 
16    * Unfortunately, that API is a ``static inline`` function, as are a 
17    * bunch of the functions it calls. The only solution seems to be to 
18    * copy those definitions and the supporting inline functions here. 
19    * 
20    * Now, this makes us VERY fragile to changes in those functions. Because 
21    * they're internal and static, the CPython devs might feel free to change 
22    * them in even minor versions, meaning that we could runtime link and load, 
23    * but still crash. We have that problem on all platforms though. It's just worse 
24    * here because we have to keep copying the updated definitions. 
25    */
26   #include <Python.h>
27   #include "greenlet_cpython_compat.hpp"

49   #define Py_TAG_BITS 0
50   #else
51   #define Py_TAG_BITS     ((uintptr_t)1)
52   #define Py_TAG_DEFERRED (1)
53   #define Py_INT_TAG      3
54   #endif
55

57   static const _PyStackRef PyStackRef_NULL = { .bits = Py_TAG_DEFERRED};
58   #define PyStackRef_IsNull(stackref) ((stackref).bits == PyStackRef_NULL.bits)
59   
60   static inline bool
61   PyStackRef_IsTaggedInt(_PyStackRef i)
62   {
63       return (i.bits & Py_INT_TAG) == Py_INT_TAG;
64   }
65   
66   static inline bool
67   PyStackRef_IsNullOrInt(_PyStackRef stackref)
68   {
69       return PyStackRef_IsNull(stackref) || PyStackRef_IsTaggedInt(stackref);
70   }
71   
72   #define _Py_VISIT_STACKREF(ref)                                         \
73       do {                                                                \
74           if (!PyStackRef_IsNullOrInt(ref)) {                             \
75               int vret = _PyGC_VisitStackRef(&(ref), visit, arg);         \
76               if (vret)                                                   \
77                   return vret;                                            \
78           }                                                               \
79       } while (0)
80   
81   static inline PyObject *
82   PyStackRef_AsPyObjectBorrow(_PyStackRef stackref)
83   {
84       PyObject *cleared = ((PyObject *)((stackref).bits & (~Py_TAG_BITS)));
85       return cleared;
86   }
87   
88   #define Py_TAG_REFCNT 1
89   #define BITS_TO_PTR(ref) ((PyObject *)((ref).bits))
90   
91   #define PyStackRef_RefcountOnObject(ref) (((ref).bits & Py_TAG_REFCNT) == 0)
92   
93   #define PyStackRef_CLOSE(REF)                            \
94       do {                                                 \
95           _PyStackRef _close_tmp = (REF);                  \
96           if (PyStackRef_RefcountOnObject(_close_tmp)) {   \
97               Py_DECREF(BITS_TO_PTR(_close_tmp));          \
98           }                                                \
99       } while (0)
100  
101  #define PyStackRef_CLEAR(REF)                            \
102      do {                                                 \
103          _PyStackRef* _clear_ptr = &(REF);                \
104          _PyStackRef _clear_old = (*_clear_ptr);          \
105          *_clear_ptr = PyStackRef_NULL;                   \
106          PyStackRef_CLOSE(_clear_old);                    \
107      } while (0)

109  static inline PyCodeObject *_PyFrame_GetCode(_PyInterpreterFrame *f) {
110      assert(!PyStackRef_IsNullOrInt(f->f_executable));
111      PyObject *executable = PyStackRef_AsPyObjectBorrow(f->f_executable);
112      assert(PyCode_Check(executable));
113      return (PyCodeObject *)executable;
114  }

116  
117  static inline _Py_CODEUNIT *
118  _PyFrame_GetBytecode(_PyInterpreterFrame *f)
119  {
120  #ifdef Py_GIL_DISABLED
121      PyCodeObject *co = _PyFrame_GetCode(f);
122      _PyCodeArray *tlbc = _PyCode_GetTLBCArray(co);
123      assert(f->tlbc_index >= 0 && f->tlbc_index < tlbc->size);
124      return (_Py_CODEUNIT *)tlbc->entries[f->tlbc_index];
125  #else
126      return _PyCode_CODE(_PyFrame_GetCode(f));
127  #endif
128  }

130  static inline bool //_Py_NO_SANITIZE_THREAD
131  _PyFrame_IsIncomplete(_PyInterpreterFrame *frame)
132  {
133      if (frame->owner >= FRAME_OWNED_BY_INTERPRETER) {
134          return true;
135      }
136      return frame->owner != FRAME_OWNED_BY_GENERATOR &&
137             frame->instr_ptr < _PyFrame_GetBytecode(frame) +
138                                    _PyFrame_GetCode(frame)->_co_firsttraceable;
139  }
