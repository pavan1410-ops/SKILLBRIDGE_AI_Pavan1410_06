8    """Horizontal sharding support. 
9     
10   Defines a rudimental 'horizontal sharding' system which allows a Session to 
11   distribute queries and persistence operations across multiple databases. 
12    
13   For a usage example, see the :ref:`examples_sharding` example included in 
14   the source distribution. 
15    
16   .. deepalchemy:: The horizontal sharding extension is an advanced feature, 
17      involving a complex statement -> database interaction as well as 
18      use of semi-public APIs for non-trivial cases.   Simpler approaches to 
19      referring to multiple database "shards", most commonly using a distinct 
20      :class:`_orm.Session` per "shard", should always be considered first 
21      before using this more complex and less-production-tested system. 
22    
23    
24    
25   """

103  class ShardedQuery(Query[_T]):
104      """Query class used with :class:`.ShardedSession`. 
105   
106      .. legacy:: The :class:`.ShardedQuery` is a subclass of the legacy 
107         :class:`.Query` class.   The :class:`.ShardedSession` now supports 
108         2.0 style execution via the :meth:`.ShardedSession.execute` method. 
109   
110      """
111  
112      def __init__(self, *args: Any, **kwargs: Any) -> None:
113          super().__init__(*args, **kwargs)
114          assert isinstance(self.session, ShardedSession)
115  
116          self.identity_chooser = self.session.identity_chooser
117          self.execute_chooser = self.session.execute_chooser
118          self._shard_id = None
119  
120      def set_shard(self, shard_id: ShardIdentifier) -> Self:
121          """Return a new query, limited to a single shard ID. 
122   
123          All subsequent operations with the returned query will 
124          be against the single shard regardless of other state. 
125   
126          The shard_id can be passed for a 2.0 style execution to the 
127          bind_arguments dictionary of :meth:`.Session.execute`:: 
128   
129              results = session.execute(stmt, bind_arguments={"shard_id": "my_shard"}) 
130   
131          """  # noqa: E501
132          return self.execution_options(_sa_shard_id=shard_id)

135  class ShardedSession(Session):
136      shard_chooser: ShardChooser
137      identity_chooser: IdentityChooser
138      execute_chooser: Callable[[ORMExecuteState], Iterable[Any]]
139  
140      def __init__(
141          self,
142          shard_chooser: ShardChooser,
143          identity_chooser: Optional[IdentityChooser] = None,
144          execute_chooser: Optional[
145              Callable[[ORMExecuteState], Iterable[Any]]
146          ] = None,
147          shards: Optional[Dict[str, Any]] = None,
148          query_cls: Type[Query[_T]] = ShardedQuery,
149          *,
150          id_chooser: Optional[
151              Callable[[Query[_T], Iterable[_T]], Iterable[Any]]
152          ] = None,
153          query_chooser: Optional[Callable[[Executable], Iterable[Any]]] = None,
154          **kwargs: Any,
155      ) -> None:
156          """Construct a ShardedSession. 
157   
158          :param shard_chooser: A callable which, passed a Mapper, a mapped 
159            instance, and possibly a SQL clause, returns a shard ID.  This id 
160            may be based off of the attributes present within the object, or on 
161            some round-robin scheme. If the scheme is based on a selection, it 
162            should set whatever state on the instance to mark it in the future as 
163            participating in that shard. 
164   
165          :param identity_chooser: A callable, passed a Mapper and primary key 
166           argument, which should return a list of shard ids where this 
167           primary key might reside. 
168   
169            .. versionchanged:: 2.0  The ``identity_chooser`` parameter 
170               supersedes the ``id_chooser`` parameter. 
171   
172          :param execute_chooser: For a given :class:`.ORMExecuteState`, 
173            returns the list of shard_ids 
174            where the query should be issued.  Results from all shards returned 
175            will be combined together into a single listing. 
176   
177            .. versionchanged:: 1.4  The ``execute_chooser`` parameter 
178               supersedes the ``query_chooser`` parameter. 
179   
180          :param shards: A dictionary of string shard names 
181            to :class:`~sqlalchemy.engine.Engine` objects. 
182   
183          """
184          super().__init__(query_cls=query_cls, **kwargs)
185  
186          event.listen(
187              self, "do_orm_execute", execute_and_instances, retval=True
188          )
189          self.shard_chooser = shard_chooser
190  
191          if id_chooser:
192              _id_chooser = id_chooser
193              util.warn_deprecated(
194                  "The ``id_chooser`` parameter is deprecated; "
195                  "please use ``identity_chooser``.",
196                  "2.0",
197              )
198  
199              def _legacy_identity_chooser(
200                  mapper: Mapper[_T],
201                  primary_key: _PKIdentityArgument,
202                  *,
203                  lazy_loaded_from: Optional[InstanceState[Any]],
204                  execution_options: OrmExecuteOptionsParameter,
205                  bind_arguments: _BindArguments,
206                  **kw: Any,
207              ) -> Any:
208                  q = self.query(mapper)
209                  if lazy_loaded_from:
210                      q = q._set_lazyload_from(lazy_loaded_from)
211                  return _id_chooser(q, primary_key)
212  
213              self.identity_chooser = _legacy_identity_chooser
214          elif identity_chooser:
215              self.identity_chooser = identity_chooser
216          else:
217              raise exc.ArgumentError(
218                  "identity_chooser or id_chooser is required"
219              )
220  
221          if query_chooser:
222              _query_chooser = query_chooser
223              util.warn_deprecated(
224                  "The ``query_chooser`` parameter is deprecated; "
225                  "please use ``execute_chooser``.",
226                  "1.4",
227              )
228              if execute_chooser:
229                  raise exc.ArgumentError(
230                      "Can't pass query_chooser and execute_chooser "
231                      "at the same time."
232                  )
233  
234              def _default_execute_chooser(
235                  orm_context: ORMExecuteState,
236              ) -> Iterable[Any]:
237                  return _query_chooser(orm_context.statement)
238  
239              if execute_chooser is None:
240                  execute_chooser = _default_execute_chooser
241  
242          if execute_chooser is None:
243              raise exc.ArgumentError(
244                  "execute_chooser or query_chooser is required"
245              )
246          self.execute_chooser = execute_chooser
247          self.__shards: Dict[ShardIdentifier, _SessionBind] = {}
248          if shards is not None:
249              for k in shards:
250                  self.bind_shard(k, shards[k])
251  
252      def _identity_lookup(
253          self,
254          mapper: Mapper[_O],
255          primary_key_identity: Union[Any, Tuple[Any, ...]],
256          identity_token: Optional[Any] = None,
257          passive: PassiveFlag = PassiveFlag.PASSIVE_OFF,
258          lazy_loaded_from: Optional[InstanceState[Any]] = None,
259          execution_options: OrmExecuteOptionsParameter = util.EMPTY_DICT,
260          bind_arguments: Optional[_BindArguments] = None,
261          **kw: Any,
262      ) -> Union[Optional[_O], LoaderCallableStatus]:
263          """override the default :meth:`.Session._identity_lookup` method so 
264          that we search for a given non-token primary key identity across all 
265          possible identity tokens (e.g. shard ids). 
266   
267          .. versionchanged:: 1.4  Moved :meth:`.Session._identity_lookup` from 
268             the :class:`_query.Query` object to the :class:`.Session`. 
269   
270          """
271  
272          if identity_token is not None:
273              obj = super()._identity_lookup(
274                  mapper,
275                  primary_key_identity,
276                  identity_token=identity_token,
277                  **kw,
278              )
279  
280              return obj
281          else:
282              for shard_id in self.identity_chooser(
283                  mapper,
284                  primary_key_identity,
285                  lazy_loaded_from=lazy_loaded_from,
286                  execution_options=execution_options,
287                  bind_arguments=dict(bind_arguments) if bind_arguments else {},
288              ):
289                  obj2 = super()._identity_lookup(
290                      mapper,
291                      primary_key_identity,
292                      identity_token=shard_id,
293                      lazy_loaded_from=lazy_loaded_from,
294                      **kw,
295                  )
296                  if obj2 is not None:
297                      return obj2
298  
299              return None

301      def _choose_shard_and_assign(
302          self,
303          mapper: Optional[_EntityBindKey[_O]],
304          instance: Any,
305          **kw: Any,
306      ) -> Any:
307          if instance is not None:
308              state = inspect(instance)
309              if state.key:
310                  token = state.key[2]
311                  assert token is not None
312                  return token
313              elif state.identity_token:
314                  return state.identity_token
315  
316          assert isinstance(mapper, Mapper)
317          shard_id = self.shard_chooser(mapper, instance, **kw)
318          if instance is not None:
319              state.identity_token = shard_id
320          return shard_id
321  
322      def connection_callable(
323          self,
324          mapper: Optional[Mapper[_T]] = None,
325          instance: Optional[Any] = None,
326          shard_id: Optional[ShardIdentifier] = None,
327          **kw: Any,
328      ) -> Connection:
329          """Provide a :class:`_engine.Connection` to use in the unit of work 
330          flush process. 
331   
332          """
333  
334          if shard_id is None:
335              shard_id = self._choose_shard_and_assign(mapper, instance)
336  
337          if self.in_transaction():
338              trans = self.get_transaction()
339              assert trans is not None
340              return trans.connection(mapper, shard_id=shard_id)
341          else:
342              bind = self.get_bind(
343                  mapper=mapper, shard_id=shard_id, instance=instance
344              )
345  
346              if isinstance(bind, Engine):
347                  return bind.connect(**kw)
348              else:
349                  assert isinstance(bind, Connection)
350                  return bind
351  
352      def get_bind(
353          self,
354          mapper: Optional[_EntityBindKey[_O]] = None,
355          *,
356          shard_id: Optional[ShardIdentifier] = None,
357          instance: Optional[Any] = None,
358          clause: Optional[ClauseElement] = None,
359          **kw: Any,
360      ) -> _SessionBind:
361          if shard_id is None:
362              shard_id = self._choose_shard_and_assign(
363                  mapper, instance=instance, clause=clause
364              )
365              assert shard_id is not None
366          return self.__shards[shard_id]
367  
368      def bind_shard(
369          self, shard_id: ShardIdentifier, bind: Union[Engine, OptionEngine]
370      ) -> None:
371          self.__shards[shard_id] = bind

374  class set_shard_id(ORMOption):
375      """a loader option for statements to apply a specific shard id to the 
376      primary query as well as for additional relationship and column 
377      loaders. 
378   
379      The :class:`_horizontal.set_shard_id` option may be applied using 
380      the :meth:`_sql.Executable.options` method of any executable statement:: 
381   
382          stmt = ( 
383              select(MyObject) 
384              .where(MyObject.name == "some name") 
385              .options(set_shard_id("shard1")) 
386          ) 
387   
388      Above, the statement when invoked will limit to the "shard1" shard 
389      identifier for the primary query as well as for all relationship and 
390      column loading strategies, including eager loaders such as 
391      :func:`_orm.selectinload`, deferred column loaders like :func:`_orm.defer`, 
392      and the lazy relationship loader :func:`_orm.lazyload`. 
393   
394      In this way, the :class:`_horizontal.set_shard_id` option has much wider 
395      scope than using the "shard_id" argument within the 
396      :paramref:`_orm.Session.execute.bind_arguments` dictionary. 
397   
398   
399      .. versionadded:: 2.0.0 
400   
401      """
402  
403      __slots__ = ("shard_id", "propagate_to_loaders")
404  
405      def __init__(
406          self, shard_id: ShardIdentifier, propagate_to_loaders: bool = True
407      ):
408          """Construct a :class:`_horizontal.set_shard_id` option. 
409   
410          :param shard_id: shard identifier 
411          :param propagate_to_loaders: if left at its default of ``True``, the 
412           shard option will take place for lazy loaders such as 
413           :func:`_orm.lazyload` and :func:`_orm.defer`; if False, the option 
414           will not be propagated to loaded objects. Note that :func:`_orm.defer` 
415           always limits to the shard_id of the parent row in any case, so the 
416           parameter only has a net effect on the behavior of the 
417           :func:`_orm.lazyload` strategy. 
418   
419          """
420          self.shard_id = shard_id
421          self.propagate_to_loaders = propagate_to_loaders

424  def execute_and_instances(
425      orm_context: ORMExecuteState,
426  ) -> Union[Result[_T], IteratorResult[_TP]]:
427      active_options: Union[
428          None,
429          QueryContext.default_load_options,
430          Type[QueryContext.default_load_options],
431          BulkUDCompileState.default_update_options,
432          Type[BulkUDCompileState.default_update_options],
433      ]
434  
435      if orm_context.is_select:
436          active_options = orm_context.load_options
437  
438      elif orm_context.is_update or orm_context.is_delete:
439          active_options = orm_context.update_delete_options
440      else:
441          active_options = None
442  
443      session = orm_context.session
444      assert isinstance(session, ShardedSession)
445  
446      def iter_for_shard(
447          shard_id: ShardIdentifier,
448      ) -> Union[Result[_T], IteratorResult[_TP]]:
449          bind_arguments = dict(orm_context.bind_arguments)
450          bind_arguments["shard_id"] = shard_id
451  
452          orm_context.update_execution_options(identity_token=shard_id)
453          return orm_context.invoke_statement(bind_arguments=bind_arguments)
454  
455      for orm_opt in orm_context._non_compile_orm_options:
456          # TODO: if we had an ORMOption that gets applied at ORM statement
457          # execution time, that would allow this to be more generalized.
458          # for now just iterate and look for our options
459          if isinstance(orm_opt, set_shard_id):
460              shard_id = orm_opt.shard_id
461              break
462      else:
463          if active_options and active_options._identity_token is not None:
464              shard_id = active_options._identity_token
465          elif "_sa_shard_id" in orm_context.execution_options:
466              shard_id = orm_context.execution_options["_sa_shard_id"]
467          elif "shard_id" in orm_context.bind_arguments:
468              shard_id = orm_context.bind_arguments["shard_id"]
469          else:
470              shard_id = None
471  
472      if shard_id is not None:
473          return iter_for_shard(shard_id)
474      else:
475          partial = []
476          for shard_id in session.execute_chooser(orm_context):
477              result_ = iter_for_shard(shard_id)
478              partial.append(result_)
479          return partial[0].merge(*partial[1:])
