313  def _normalize_skill_name(skill):
314  
315      if skill is None:
316          return ""
317  
318      return " ".join(
319          str(skill)
320          .strip()
321          .lower()
322          .split()
323      )

327  # SAFE NUMBER
328  # =========================================================
329  
330  def _safe_number(value, default=0):
331  
332      try:
333          return float(value)
334      except (TypeError, ValueError):
335          return default

339  # GET LEARNING EVIDENCE
340  # =========================================================
341  
342  def get_learning_evidence(student):
343  
344      """ 
345      Returns the highest saved learning progress for 
346      each skill. 
347   
348      Example: 
349   
350          Communication -> 100 
351          Python        -> 60 
352          SQL           -> 25 
353      """
354  
355      evidence = {}
356  
357      if not student:
358          return evidence
359  
360      try:
361          progress_items = get_student_learning_progress(student)
362      except Exception:
363          progress_items = []
364  
365      for item in progress_items:
366  
367          skill = _normalize_skill_name(
368              getattr(item, "skill", "")
369          )
370  
371          if not skill:
372              continue
373  
374          progress = _safe_number(
375              getattr(
376                  item,
377                  "progress_percentage",
378                  0
379              )
380          )
381  
382          progress = max(
383              0,
384              min(100, progress)
385          )
386  
387          current = evidence.get(
388              skill,
389              0
390          )
391  
392          if progress > current:
393              evidence[skill] = progress
394  
395      return evidence

402  def get_learning_duration(gap):
403  
404      gap = _safe_number(gap)
405  
406      if gap >= 51:
407          return "6-8 weeks"
408  
409      elif gap >= 26:
410          return "4-6 weeks"
411  
412      elif gap >= 11:
413          return "2-4 weeks"
414  
415      elif gap > 0:
416          return "1-2 weeks"
417  
418      return "Completed"

467  # =========================================================
468  
469  def get_prioritized_gaps(student, career):
470  
471      if not student or not career:
472          return []
473  
474      gaps = calculate_skill_gaps(
475          student,
476          career
477      )
478  
479      if not gaps:
480          return []
481  
482      priority_order = {
483          "Critical": 1,
484          "High": 2,
485          "Medium": 3,
486          "Low": 4
487      }
488  
489      gaps.sort(
490          key=lambda item: (
491              priority_order.get(
492                  item.get("priority"),
493                  5
494              ),
495              -_safe_number(
496                  item.get("gap", 0)
497              )
498          )
499      )
500  
501      return gaps

505  # GENERATE LEARNING PATH
506  # =========================================================
507  
508  def generate_learning_path(student, career):
509  
510      if not student or not career:
511          return []
512  
513      gaps = get_prioritized_gaps(
514          student,
515          career
516      )
517  
518      # -----------------------------------------------------
519      # IMPORTANT
520      #
521      # Only skills with an actual remaining gap are
522      # included.
523      #
524      # If Communication becomes:
525      #
526      # Current  = 100
527      # Required = 70
528      # Gap      = 0
529      #
530      # it will NOT appear here.
531      # -----------------------------------------------------
532  
533      gaps = [
534  
535          item
536  
537          for item in gaps
538  
539          if _safe_number(
540              item.get("gap", 0)
541          ) > 0
542  
543      ]
544  
545      learning_path = []
546  
547      learning_evidence = get_learning_evidence(
548          student
549      )
550  
551      for index, item in enumerate(
552          gaps,
553          start=1
554      ):
555  
556          skill = str(
557              item.get(
558                  "skill",
559                  "Unknown Skill"
560              )
561          ).strip()
562  
563          current = _safe_number(
564              item.get(
565                  "current",
566                  0
567              )
568          )
569  
570          required = _safe_number(
571              item.get(
572                  "required",
573                  0
574              )
575          )
576  
577          gap = _safe_number(
578              item.get(
579                  "gap",
580                  0
581              )
582          )
583  
584          # -------------------------------------------------
585          # LEARNING EVIDENCE
586          # -------------------------------------------------
587  
588          evidence = learning_evidence.get(
589              _normalize_skill_name(skill),
590              0
591          )
592  
593          # The skill-gap service should already include
594          # learning evidence. This extra check prevents
595          # stale values from producing an incorrect path.
596          if evidence > current:
597              current = evidence
598  
599          # Recalculate remaining gap safely.
600          gap = max(
601              0,
602              required - current
603          )
604  
605          # If learning evidence has closed the gap,
606          # do not add the skill to the path.
607          if gap <= 0:
608              continue
609  
610          priority = item.get(
611              "priority"
612          )
613  
614          # Recalculate priority using the effective gap.
615          priority = get_learning_priority(
616              gap
617          )
618  
619          duration = get_learning_duration(
620              gap
621          )
622  
623          resource = get_learning_resource(
624              skill
625          )
626  
627          learning_path.append({
628  
629              "step": index,
630  
631              "skill": skill,
632  
633              "current": int(
634                  round(current)
635              ),
636  
637              "required": int(
638                  round(required)
639              ),
640  
641              "gap": int(
642                  round(gap)
643              ),
644  
645              "percentage": item.get(
646                  "percentage",
647                  0
648              ),
649  
650              "status": item.get(
651                  "status",
652                  "Needs Improvement"
653              ),
654  
655              "priority": priority,
656  
657              "duration": duration,
658  
659              "course": resource.get(
660                  "course",
661                  f"{skill} Fundamentals"
662              ),
663  
664              "topics": resource.get(
665                  "topics",
666                  []
667              ),
668  
669              "learning_status": "Not Started",
670  
671              "progress_percentage": int(
672                  round(evidence)
673              )
674  
675          })
676  
677      # Re-number after any completed skills were removed.
678      for index, item in enumerate(
679          learning_path,
680          start=1
681      ):
682          item["step"] = index
683  
684      return learning_path

784  def get_learning_summary(student, career):
785  
786      path = generate_learning_path(
787          student,
788          career
789      )
790  
791      if not path:
792  
793          return {
794              "total_steps": 0,
795              "critical": 0,
796              "high": 0,
797              "medium": 0,
798              "low": 0,
799              "message": (
800                  "You have completed all required "
801                  "skills!"
802              )
803          }
804  
805      return {
806  
807          "total_steps": len(path),
808  
809          "critical": sum(
810              1
811              for item in path
812              if item.get(
813                  "priority"
814              ) == "Critical"
815          ),
816  
817          "high": sum(
818              1
819              for item in path
820              if item.get(
821                  "priority"
822              ) == "High"
823          ),
824  
825          "medium": sum(
826              1
827              for item in path
828              if item.get(
829                  "priority"
830              ) == "Medium"
831          ),
832  
833          "low": sum(
834              1
835              for item in path
836              if item.get(
837                  "priority"
838              ) == "Low"
839          ),
840  
841          "message": (
842              f"{len(path)} skills need improvement "
843              f"for your {career} career goal."
844          )
845      }
